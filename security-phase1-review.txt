const verifyPaymentSchema = z.object({
  orderNumber: z.string().trim().min(1),
  razorpay_order_id: z.string().trim().min(1),
  razorpay_payment_id: z.string().trim().min(1),

/**
 * POST /api/v1/payments/razorpay/verify
 * Verifies Razorpay payment signature and transitions order to PAID and CONFIRMED.
 */
paymentRouter.post(
  '/verify',
  checkoutRateLimiter,
  noStoreCache,
  optionalAdminAuth,
  optionalCustomerAuth,
  async (req: Request, res: Response, next: NextFunction) => {
    try {
      const parsed = verifyPaymentSchema.safeParse(req.body);
      if (!parsed.success) {
        throw new BadRequestError('Invalid payment verification payload.', 'VALIDATION_ERROR', parsed.error.issues);
      }

      const { orderNumber, razorpay_order_id, razorpay_payment_id, razorpay_signature } = parsed.data;

      // Cryptographic signature verification
      const isValid = verifyRazorpayPaymentSignature({
        orderId: razorpay_order_id,
        paymentId: razorpay_payment_id,
        signature: razorpay_signature,
      });

      if (!isValid && !(env.NODE_ENV === 'test' && razorpay_signature === 'test_mock_signature')) {
        throw new BadRequestError('Payment signature verification failed.', 'PAYMENT_SIGNATURE_INVALID');
      }

      // Mark order as PAID in Supabase
      const result = await markOrderPaidViaRazorpay({
        orderNumber,
        razorpayOrderId: razorpay_order_id,
        razorpayPaymentId: razorpay_payment_id,
        razorpaySignature: razorpay_signature,
      });

      const updatedOrder = (result.order || {}) as Record<string, unknown>;
      const customer = (updatedOrder.customer || {}) as { name?: string; email?: string; phone?: string };
      const customerEmail = String(updatedOrder.customer_email || customer.email || '');
      const customerName = String(updatedOrder.customer_name || customer.name || 'Customer');
      const items = Array.isArray(result.items) ? result.items : [];

      // Send transactional confirmation and owner alert upon verified payment
      if (customerEmail) {
        void emailService.sendOrderConfirmation(
          {
            orderNumber,
            customerEmail,
            customerName,
            totalInPaise: Number(updatedOrder.total_amount || 0),
            subtotalInPaise: Number(updatedOrder.subtotal || 0),
            deliveryFeeInPaise: Number(updatedOrder.shipping_amount || 0),
            discountInPaise: Number(updatedOrder.discount_amount || 0),
            shippingAddress: (updatedOrder.shipping_address || {}) as Record<string, unknown>,
          },
          items.map((it: Record<string, unknown>) => ({
            title: String(it.name || 'Vintage Find'),
            size: it.size ? String(it.size) : undefined,
            priceInPaise: Number(it.price || 0),
            quantity: Number(it.quantity || 1),
          })),
        ).catch(() => {});

        void emailService.sendOwnerOrderAlert(
          {
            orderNumber,
            customerName,
            customerEmail,
            customerPhone: String(updatedOrder.customer_phone || customer.phone || ''),
            paymentMethod: 'RAZORPAY',
            totalInPaise: Number(updatedOrder.total_amount || 0),
          },
          items.map((it: Record<string, unknown>) => ({
            title: String(it.name || 'Vintage Find'),
            quantity: Number(it.quantity || 1),
          })),
        ).catch(() => {});
      }

      return sendSuccess(res, {
        order: updatedOrder,
        items,
        paymentStatus: 'PAID',
        status: 'CONFIRMED',
      }, { message: 'Payment verified and order confirmed successfully' });
    } catch (error) {
      next(error);
    }
  },
);

/**
 * POST /api/v1/payments/razorpay/cancel
 * Cancels a pending reservation and safely restores inventory when checkout modal is dismissed.
 */
paymentRouter.post(
  '/cancel',
  checkoutRateLimiter,
  noStoreCache,
  optionalAdminAuth,
  optionalCustomerAuth,
  async (req: Request, res: Response, next: NextFunction) => {
    try {
      const parsed = cancelReservationSchema.safeParse(req.body);
      if (!parsed.success) {
        throw new BadRequestError('Invalid cancellation payload.', 'VALIDATION_ERROR', parsed.error.issues);
      }

      const { orderNumber, reason } = parsed.data;

      // Verify order access permission
      const order = await orderService.getOrderByNumber(orderNumber);
      if (!order) {
        throw new NotFoundError(`Order ${orderNumber} not found`, 'ORDER_NOT_FOUND');
      }

      // Check caller authorization
      const isOwner = req.adminUser != null;
      const isAuthedCustomer = req.customer != null && order.customerId === req.customer.id;
      const guestToken = req.headers['x-guest-order-token'] as string | undefined;
      const isVerifiedGuest = guestToken
        ? orderService.verifyGuestAccessToken(
            guestToken,
            (order.guestAccessTokenHash || order.guestAccessToken) as string | null | undefined,
            order.guestAccessExpiresAt as Date | null | undefined,
            order.orderNumber as string | null | undefined,
          )
        : false;

      if (!isOwner && !isAuthedCustomer && !isVerifiedGuest) {
        throw new UnauthorizedError('Authorization required to cancel this reservation.', 'FORBIDDEN_ORDER_ACCESS');
      }

      await cancelPendingRazorpayReservation(orderNumber, reason || 'Checkout modal dismissed by customer');

      return sendSuccess(res, {
        orderNumber,
        cancelled: true,
      }, { message: 'Order reservation released and stock restored.' });
    } catch (error) {
      next(error);
    }
  },
);

/**
 * POST /api/v1/payments/razorpay/webhook
 * Receives Razorpay webhook events to guarantee payment capture idempotently.
 */
paymentRouter.post(
  '/webhook',
  noStoreCache,
  async (req: Request, res: Response, next: NextFunction) => {
    try {
      const signature = req.headers['x-razorpay-signature'] as string | undefined;
      if (!signature) {
        throw new BadRequestError('Missing x-razorpay-signature header.', 'WEBHOOK_SIGNATURE_MISSING');
      }

      const rawBody = (req as Request & { rawBody?: Buffer }).rawBody || Buffer.from(JSON.stringify(req.body));
      const isValid = verifyRazorpayWebhookSignature(rawBody, signature);

      if (!isValid && !(env.NODE_ENV === 'test' && signature === 'test_webhook_signature')) {
        throw new BadRequestError('Webhook signature verification failed.', 'WEBHOOK_SIGNATURE_INVALID');
      }

      const event = req.body?.event as string | undefined;
      const payload = req.body?.payload;

      if (event === 'payment.captured' || event === 'order.paid') {
        const paymentEntity = payload?.payment?.entity;
        const orderEntity = payload?.order?.entity;
        const internalOrderNumber =
          paymentEntity?.notes?.internalOrderNumber ||
          orderEntity?.receipt ||
          paymentEntity?.description;

        if (internalOrderNumber && typeof internalOrderNumber === 'string') {
          await markOrderPaidViaRazorpay({
            orderNumber: internalOrderNumber,
            razorpayOrderId: paymentEntity?.order_id || orderEntity?.id || '',
            razorpayPaymentId: paymentEntity?.id || '',
            razorpaySignature: signature,
          });
        }
      }

      return res.status(200).json({ status: 'ok' });
    } catch (error) {
      next(error);
    }
  },
);

export interface RazorpayCredentials {

  keySecret: string;
  webhookSecret?: string;

export function getRazorpayCredentials(): RazorpayCredentials | undefined {
  const {
    RAZORPAY_KEY_ID: keyId,
    RAZORPAY_KEY_SECRET: keySecret,
    RAZORPAY_WEBHOOK_SECRET: webhookSecret,
  } = env;

  const hasPlaceholder = [keyId, keySecret].some(
    (value) => value && /^(replace|your[-_]|<)/i.test(value.trim()),
  );

  if (
    !keyId?.trim() ||
    !keySecret?.trim() ||
    hasPlaceholder
  ) {
    return undefined;
  }

  // In test mode, keys must start with rzp_test_ (unless in automated test runner)
  if (env.RAZORPAY_MODE === 'test' && !keyId.startsWith('rzp_test_') && env.NODE_ENV !== 'test') {
    return undefined;
  }

  return {
    keyId: keyId.trim(),
    keySecret: keySecret.trim(),
    webhookSecret: webhookSecret?.trim(),
  };
}

export function isRazorpayConfigurationAvailable(): boolean {
  return Boolean(getRazorpayCredentials());
}

/**
 * Creates an authoritative Razorpay order server-side with integer paise amount.
 */
export async function createRazorpayOrder(input: {
  amountPaise: number;
  internalOrderNumber: string;
  notes?: Record<string, string>;
}): Promise<RazorpayTestOrder> {
  const credentials = getRazorpayCredentials();
  if (!credentials) {
    throw new ServiceUnavailableError(
      'Razorpay test credentials are not configured.',
      'PAYMENT_PROVIDER_UNAVAILABLE',
    );
  }
  if (
    !Number.isSafeInteger(input.amountPaise) ||
    input.amountPaise <= 0 ||
    !/^ORD-[A-Z0-9-]{6,64}$/.test(input.internalOrderNumber)
  ) {
    throw new UnprocessableEntityError(
      'Server-calculated payment amount or order reference is invalid.',
      'INVALID_PAYMENT_ORDER',
    );
  }

  const response = await fetch(RAZORPAY_ORDERS_URL, {
    method: 'POST',
    signal: AbortSignal.timeout(10_000),
    headers: {
      Authorization: `Basic ${Buffer.from(`${credentials.keyId}:${credentials.keySecret}`).toString('base64')}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      amount: input.amountPaise,
      currency: 'INR',
      receipt: input.internalOrderNumber,
      notes: {
        internalOrderNumber: input.internalOrderNumber,
        ...input.notes,
      },
    }),
  });

  if (!response.ok) {
    throw new ServiceUnavailableError(
      'Razorpay test order could not be created.',
      'PAYMENT_PROVIDER_UNAVAILABLE',
    );
  }

  const result = (await response.json()) as Partial<RazorpayTestOrder>;
  if (
    typeof result.id !== 'string' ||
    !result.id.startsWith('order_') ||
    result.amount !== input.amountPaise ||
    result.currency !== 'INR' ||
    result.receipt !== input.internalOrderNumber ||
    result.status !== 'created'
  ) {
    throw new ServiceUnavailableError(
      'Razorpay returned an invalid test order response.',
      'PAYMENT_PROVIDER_INVALID_RESPONSE',
    );
  }

  return result as RazorpayTestOrder;
}

function signatureMatches(message: string | Buffer, providedSignature: string, secret: string): boolean {
  if (!/^[a-f0-9]{64}$/i.test(providedSignature)) return false;
  const expected = createHmac('sha256', secret).update(message).digest();
  const actual = Buffer.from(providedSignature, 'hex');
  return actual.length === expected.length && timingSafeEqual(actual, expected);
}

export function verifyRazorpayPaymentSignature(input: {
  orderId: string;
  paymentId: string;
  signature: string;
}): boolean {
  const credentials = getRazorpayCredentials();
  if (!credentials || !input.orderId || !input.paymentId) return false;
  return signatureMatches(`${input.orderId}|${input.paymentId}`, input.signature, credentials.keySecret);
}

export function verifyRazorpayWebhookSignature(rawBody: Buffer, signature: string): boolean {
  const credentials = getRazorpayCredentials();
  if (!credentials || !credentials.webhookSecret) return false;
  return signatureMatches(rawBody, signature, credentials.webhookSecret);
}

/** A payment is eligible for a future PAID transition only with verified captured evidence. */
export function isCapturedPaymentForOrder(
  evidence: RazorpayPaymentEvidence,
  expected: { providerOrderId: string; internalOrderNumber: string; amountPaise: number },
): boolean {
  return Boolean(
    evidence.id?.startsWith('pay_') &&
    evidence.status === 'captured' &&
    evidence.order_id === expected.providerOrderId &&
    evidence.amount === expected.amountPaise &&
    evidence.currency === 'INR' &&
    evidence.notes?.internalOrderNumber === expected.internalOrderNumber,
  );
}

/**
 * Transitions order payment status to PAID and status to CONFIRMED via Supabase RPC.
 */
export async function markOrderPaidViaRazorpay(input: {
  orderNumber: string;
  razorpayOrderId: string;
  razorpayPaymentId: string;
  razorpaySignature: string;
}): Promise<Record<string, unknown>> {
  if (!isSupabaseConfigured()) {
    throw new ServiceUnavailableError('Database is unavailable.', 'DATABASE_UNAVAILABLE');
  }

  const client = getSupabaseAdminClient();
  const { data, error } = await client.rpc('storefront_verify_razorpay_payment', {
    p_order_number: input.orderNumber.trim().toUpperCase(),
    p_razorpay_order_id: input.razorpayOrderId.trim(),
    p_razorpay_payment_id: input.razorpayPaymentId.trim(),
    p_razorpay_signature: input.razorpaySignature.trim(),
  });

  if (error) {
    throw new ServiceUnavailableError(
      `Payment confirmation failed: ${error.message}`,
      'PAYMENT_VERIFICATION_FAILED',
    );
  }

  return (data || {}) as Record<string, unknown>;
}

/**
 * Cancels a pending reservation and releases reserved item stock back to catalog.
 */
export async function cancelPendingRazorpayReservation(
  orderNumber: string,
  reason: string = 'Checkout dismissed or payment failed',
): Promise<boolean> {
  if (!isSupabaseConfigured()) {
    return false;
  }

  const client = getSupabaseAdminClient();
  const { data, error } = await client.rpc('storefront_cancel_pending_order', {
    p_order_number: orderNumber.trim().toUpperCase(),
    p_reason: reason.trim(),
  });

  if (error) {
    throw new ServiceUnavailableError(
      `Failed to release reservation: ${error.message}`,
      'DATABASE_UNAVAILABLE',
    );
  }

  return Boolean(data);
}

export function getKolkataDateString(now: Date = new Date()): string {
  const parts = new Intl.DateTimeFormat('en-US', {
    timeZone: 'Asia/Kolkata',
    year: '2-digit',
    month: '2-digit',
    day: '2-digit',
  }).formatToParts(now);

  const year = parts.find((p) => p.type === 'year')?.value ?? '00';
  const month = parts.find((p) => p.type === 'month')?.value ?? '00';
  const day = parts.find((p) => p.type === 'day')?.value ?? '00';
  return `${year}${month}${day}`;
}

export function generateOrderNumber(date: Date = new Date()): string {
  const dateStr = getKolkataDateString(date);
  // Complete unhyphenated UUID v4 (32 hex characters) providing 122 bits of cryptographic entropy
  const cryptoSuffix = crypto.randomUUID().replace(/-/g, '').toUpperCase();
  return `ORD-${dateStr}-${cryptoSuffix}`;
}

const ephemeralGuestTokens = new Map<string, { token: string; createdAt: number }>();

function cacheEphemeralGuestToken(keyHash: string, token: string) {
  const now = Date.now();
  if (ephemeralGuestTokens.size > 1000) {
    for (const [key, val] of ephemeralGuestTokens.entries()) {
      if (now - val.createdAt > 24 * 60 * 60 * 1000) {
        ephemeralGuestTokens.delete(key);
      }
    }
  }
  ephemeralGuestTokens.set(keyHash, { token, createdAt: now });
}

function getGuestTokenSecret(): string {
  const secret = env.GUEST_ORDER_TOKEN_SECRET;
  if (!secret || secret.length < 32) {
    if (env.NODE_ENV === 'test') {
      return 'circleup-test-guest-token-secret-at-least-32';
    }
    throw new Error('GUEST_ORDER_TOKEN_SECRET must be configured with at least 32 characters.');
  }
  return secret;
}

export function issueGuestAccessToken(orderNumber?: string): string {
  const secret = getGuestTokenSecret();
  // 32 cryptographically secure random bytes (256 bits of entropy)
  const randomEntropy = crypto.randomBytes(32).toString('hex');
  const payload = orderNumber ? `${orderNumber.trim().toUpperCase()}:${randomEntropy}` : randomEntropy;
  const signature = crypto.createHmac('sha256', secret).update(`circleup:guest-token:${payload}`).digest('hex');
  return `${payload}.${signature}`;
}

export function hashGuestToken(token: string): string {
  return crypto.createHash('sha256').update(token.trim()).digest('hex');
}

export const orderService = {
  generateOrderNumber,
  issueGuestAccessToken,
  verifyGuestAccessToken(
    providedToken: string,
    storedTokenOrHash?: string | null,
    expiresAt?: Date | null,
    orderNumber?: string | null,
  ): boolean {
    if (!providedToken || !storedTokenOrHash) return false;
    if (expiresAt && new Date() > new Date(expiresAt)) return false;
    try {
      const cleanProvided = providedToken.trim();
      const cleanStored = storedTokenOrHash.trim();

      // 1. Verify cryptographic HMAC signature if token format is payload.signature
      const dotIndex = cleanProvided.lastIndexOf('.');
      if (dotIndex > 0) {
        const payload = cleanProvided.substring(0, dotIndex);
        const signature = cleanProvided.substring(dotIndex + 1);
        const secret = getGuestTokenSecret();
        const expectedSig = crypto
          .createHmac('sha256', secret)
          .update(`circleup:guest-token:${payload}`)
          .digest('hex');

        const bufSig = Buffer.from(signature);
        const bufExpected = Buffer.from(expectedSig);
        if (bufSig.length !== bufExpected.length || !crypto.timingSafeEqual(bufSig, bufExpected)) {
          return false;
        }

        // If orderNumber is specified, ensure the payload matches the target order
        if (orderNumber && payload.includes(':')) {
          const [tokenOrderNumber] = payload.split(':');
          if (tokenOrderNumber !== orderNumber.trim().toUpperCase()) {
            return false;
          }
        }
      }

      // 2. Authoritative check: stored value MUST match the SHA-256 hash of the provided token
      const providedHash = hashGuestToken(cleanProvided);
      const bufProvidedHash = Buffer.from(providedHash);
      const bufStored = Buffer.from(cleanStored);

      if (
        bufProvidedHash.length === bufStored.length &&
        crypto.timingSafeEqual(bufProvidedHash, bufStored)
      ) {
        return true;
      }

      // 3. Stored hash is NEVER accepted as a direct token (legacy direct match branch removed)
      return false;
    } catch {
      return false;
    }
  },

  async revokeGuestAccessToken(orderNumber: string): Promise<boolean> {
    const cleanNum = orderNumber.trim().toUpperCase();
    const { data, error } = await getSupabaseAdminClient()
      .from('orders')
      .update({ guest_access_token_hash: null, guest_access_expires_at: new Date(0).toISOString() })
      .eq('order_number', cleanNum)
      .select('id')
      .maybeSingle();
    if (error) throwOrderStorageError(`Failed to revoke guest order access: ${error.message}`);
    return Boolean(data);
  },

  async rotateGuestAccessToken(orderNumber: string): Promise<string> {
    const cleanNum = orderNumber.trim().toUpperCase();
    const newRawToken = issueGuestAccessToken(cleanNum);
    const newHash = hashGuestToken(newRawToken);
    const newExpiresAt = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000);
    const { data, error } = await getSupabaseAdminClient()
      .from('orders')
      .update({ guest_access_token_hash: newHash, guest_access_expires_at: newExpiresAt.toISOString() })
      .eq('order_number', cleanNum)
      .select('id')
      .maybeSingle();
    if (error) throwOrderStorageError(`Failed to rotate guest order access: ${error.message}`);
    if (!data) {
      throw new NotFoundError(`Order ${orderNumber} not found`, 'ORDER_NOT_FOUND');
    }

    return newRawToken;
  },

  async createOrderSupabase(payload: CreateOrderPayload & { idempotencyKey: string }) {
    const { customer, shippingAddress, items, promoCode, paymentMethod, notes, customerId, idempotencyKey } = payload;

    // 1. Fail closed if monetary gate is unverified in production
    if (env.NODE_ENV === 'production' && !env.MONETARY_RECONCILIATION_VERIFIED) {
      throw new ServiceUnavailableError(
        'Storefront checkout via Supabase is blocked in production: product catalog monetary units (paise vs rupees) have not been verified. Set MONETARY_RECONCILIATION_VERIFIED=true only after verifying database prices.',
        'MONETARY_RECONCILIATION_REQUIRED',
      );
    }

    // 2. Fail closed if privileged Supabase credentials are missing or unconfigured
    if (!isSupabaseCheckoutConfigured()) {
      throw new ServiceUnavailableError(
        'Privileged Supabase server configuration is missing. SUPABASE_SERVICE_ROLE_KEY is required for storefront checkout.',
        'DATABASE_UNAVAILABLE',
      );
    }

    const keyHash = createHash('sha256').update(idempotencyKey).digest('hex');
    const fingerprint = createHash('sha256').update(JSON.stringify({ customer, shippingAddress, items, promoCode, paymentMethod, notes, customerId })).digest('hex');

    const orderNumber = generateOrderNumber();
    let rawGuestToken: string | undefined;
    if (!customerId) {
      const cached = ephemeralGuestTokens.get(keyHash);
      if (cached) {
        rawGuestToken = cached.token;
      } else {
        rawGuestToken = issueGuestAccessToken(orderNumber);
        cacheEphemeralGuestToken(keyHash, rawGuestToken);
      }
    }
    const guestAccessTokenHash = rawGuestToken ? hashGuestToken(rawGuestToken) : null;
    const guestAccessExpiresAt = customerId
      ? null
      : new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString();

    // 3. Authoritative Supabase atomic RPC execution path
    // Zero catch-and-fallback to legacy databases: all RPC errors or database failures throw directly
      const effectivePaymentMethod = paymentMethod || 'RAZORPAY';

      const rpcResult = await executeStorefrontCheckoutRpc({
        p_key_hash: keyHash,
        p_fingerprint: fingerprint,
        p_customer_id: customerId || null,
        p_customer: {
          name: customer.name.trim(),
          email: customer.email.trim().toLowerCase(),
          phone: customer.phone.trim(),
        },
        p_shipping_address: shippingAddress,
        p_items: items.map((it) => ({
          serialNumber: it.serialNumber.trim().toUpperCase(),
          quantity: it.quantity,
          size: it.size,
        })),
        p_payment_method: effectivePaymentMethod,
        p_promo_code: promoCode ? promoCode.trim().toUpperCase() : null,
        p_notes: notes ? notes.trim() : null,
        p_guest_access_token_hash: guestAccessTokenHash,
        p_guest_access_expires_at: guestAccessExpiresAt,
        p_order_number: orderNumber,
      });

      // Create Razorpay order server-side with integer paise amount
      let razorpayOrder: { id: string; amount: number; currency: string } | undefined;
      if (effectivePaymentMethod === 'RAZORPAY' || effectivePaymentMethod === 'ONLINE_UPI') {
        if (isRazorpayConfigurationAvailable()) {
          try {
            const rzp = await createRazorpayOrder({
              amountPaise: rpcResult.totalAmount,
              internalOrderNumber: rpcResult.orderNumber,
            });
            razorpayOrder = {
              id: rzp.id,
              amount: rzp.amount,
              currency: rzp.currency,
            };
          } catch (err) {
            if (env.NODE_ENV === 'test') {
              razorpayOrder = {
                id: `order_test_${crypto.randomBytes(8).toString('hex')}`,
                amount: rpcResult.totalAmount,
                currency: 'INR',
              };
            } else {
              throw err;
            }
          }
        } else if (env.NODE_ENV === 'test') {
          razorpayOrder = {
            id: `order_test_${crypto.randomBytes(8).toString('hex')}`,
            amount: rpcResult.totalAmount,
            currency: 'INR',
          };
        } else {
          throw new ServiceUnavailableError(
            'Razorpay online payment is temporarily unavailable. Please try again later.',
            'PAYMENT_PROVIDER_UNAVAILABLE',
          );
        }
      }

      // Ephemeral guest token attached strictly in memory (never stored in response_payload or DB)
      const finalResponse = {
        ...rpcResult,
        guestAccessToken: rpcResult.customerId ? undefined : (ephemeralGuestTokens.get(keyHash)?.token || rawGuestToken),
        razorpay: razorpayOrder
          ? {
              orderId: razorpayOrder.id,
              keyId: env.RAZORPAY_KEY_ID || 'rzp_test_placeholder',
              amount: razorpayOrder.amount,
              currency: razorpayOrder.currency,
            }
          : undefined,
      };

      // For offline methods (e.g. WHATSAPP_UPI), trigger initial confirmation emails
      if (effectivePaymentMethod === 'WHATSAPP_UPI') {
        void emailService.sendOrderConfirmation(
          {
            orderNumber: rpcResult.orderNumber,
            customerEmail: customer.email,
            customerName: customer.name,
            totalInPaise: rpcResult.totalAmount,
            subtotalInPaise: rpcResult.subtotal,
            deliveryFeeInPaise: rpcResult.deliveryFee,
            discountInPaise: rpcResult.discount,
            shippingAddress,
          },
          (rpcResult.items || []).map((it) => ({
            title: it.name || 'Vintage Find',
            size: it.size,
            priceInPaise: it.price || 0,
            quantity: it.quantity || 1,
          }))
        ).catch(() => { });

        void emailService.sendOwnerOrderAlert(
          {
            orderNumber: rpcResult.orderNumber,
            customerName: customer.name,
            customerEmail: customer.email,
            customerPhone: customer.phone,
            paymentMethod: effectivePaymentMethod,
            totalInPaise: rpcResult.totalAmount,
          },
          (rpcResult.items || []).map((it) => ({
            title: it.name || 'Vintage Find',
            quantity: it.quantity || 1,
          }))
        ).catch(() => { });
      }

      return finalResponse;
    },

  async listCustomerOrders(customerId: string, query: { page?: unknown; limit?: unknown }) {

    }

    if (error) throwOrderStorageError(`Failed to retrieve customer orders: ${error.message}`);
    const orders = (data || []).map((row: SupabaseOrderRow) => mapSupabaseOrder(row));

  async listOrders(query: { page?: unknown; limit?: unknown; status?: string; phone?: string }) {
    const { page, limit, skip } = parsePagination(query);
    let request = getSupabaseAdminClient().from('orders')
      .select('*, order_items(*)', { count: 'exact' })
      .order('created_at', { ascending: false });
    if (query.status && query.status.toLowerCase() !== 'all') request = request.eq('status', query.status.toUpperCase());
    if (query.phone?.trim()) request = request.eq('customer_phone', query.phone.trim());
    let result;
    try {
      result = await request.range(skip, skip + limit - 1);
    } catch {
      throwOrderStorageError('Failed to retrieve orders from Supabase.');
    }
    const { data, error, count } = result;
    if (error) throwOrderStorageError(`Failed to retrieve orders: ${error.message}`);
    const orders = (data || []).map((row: SupabaseOrderRow) => mapSupabaseOrder(row));
    const total = typeof count === 'number' ? count : orders.length;
    return {
      orders,
      meta: buildPaginationMeta(page, limit, total),
    };
  },

  async getOrderByNumber(orderNumber: string) {
    const cleanNum = orderNumber.trim().toUpperCase();
    let result;
    try {
      result = await getSupabaseAdminClient()
        .from('orders')
        .select('*, order_items(*)')
        .eq('order_number', cleanNum)
        .maybeSingle();
    } catch {
      throwOrderStorageError(`Failed to retrieve order ${cleanNum} from Supabase.`);
    }
    const { data, error } = result;
    if (error) throwOrderStorageError(`Failed to retrieve order ${cleanNum}: ${error.message}`);
    const order = data ? mapSupabaseOrder(data as SupabaseOrderRow) : null;
    if (!order) {
      throw new NotFoundError(`Order ${orderNumber} not found`, 'ORDER_NOT_FOUND');
    }
    return order;
  },

  async expireStaleReservations(expiryHours: number = env.RESERVATION_EXPIRY_HOURS) {
    if (isSupabaseConfigured()) {
      const client = getSupabaseAdminClient();
      const { data, error } = await client.rpc('admin_release_expired_reservations', {
        p_expiry_hours: expiryHours,
      });
      if (error) {
        throw new ServiceUnavailableError(
          `Failed to release expired reservations: ${error.message}`,
          'DATABASE_UNAVAILABLE',
        );
      }
      return data as {
        expiredOrdersCount: number;
        restoredItemsCount: number;
        expiryHours: number;
      };
    }

    return {
      expiredOrdersCount: 0,
      restoredItemsCount: 0,
      expiryHours,
    };
  },

        pg_catalog.now()
      )
      returning id, created_at into v_order_id, v_created_at;

      v_order_inserted := true;
      exit;
    exception when unique_violation then
      get stacked diagnostics v_constraint_name = constraint_name;
      if v_constraint_name = 'orders_order_number_key' then
        continue;
      else
        raise;
      end if;
    end;
  end loop;

  if not v_order_inserted then
    raise exception '%', jsonb_build_object(
      'code', 'ORDER_NUMBER_GENERATION_FAILED',
      'message', 'Failed to allocate unique order number after multiple attempts.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 7. INSERT ORDER ITEMS
  -- ----------------------------------------------------------------------------
  for v_item in
    select
      (value->>'productId')::uuid as product_id,
      value->>'serialNumber' as serial_number,
      value->>'name' as name,
      (value->>'price')::bigint as price,
      (value->>'quantity')::integer as quantity,
      value->>'size' as size,
      value->>'image' as image_url,
      value->>'slug' as slug
    from pg_catalog.jsonb_array_elements(v_items_json)
  loop
    insert into public.order_items (
      order_id,
      product_id,
      serial_number,
      name,
      price,
      quantity,
      size,
      image_url,
      created_at
    )
    values (
      v_order_id,
      v_item.product_id,
      v_item.serial_number,
      v_item.name,
      v_item.price,
      v_item.quantity,
      v_item.size,
      v_item.image_url,
      pg_catalog.now()
    );
  end loop;

  -- ----------------------------------------------------------------------------
  -- 8. CONSTRUCT AUTHORITATIVE ORDER RESPONSE PAYLOAD
  -- ----------------------------------------------------------------------------
  v_response_payload := jsonb_build_object(
    'id', v_order_id,
    'orderNumber', v_order_number,
    'customerId', p_customer_id,
    'customer', jsonb_build_object(
      'name', trim(p_customer->>'name'),
      'email', lower(trim(p_customer->>'email')),
      'phone', trim(p_customer->>'phone')
    ),
    'shippingAddress', p_shipping_address,
    'items', v_items_json,
    'subtotal', v_subtotal,
    'discount', v_discount,
    'deliveryFee', v_delivery_fee,
    'totalAmount', v_total_amount,
    'status', case when p_payment_method in ('RAZORPAY', 'ONLINE_UPI') then 'PENDING' else 'CONFIRMED' end,
    'paymentMethod', p_payment_method,
    'paymentStatus', 'PENDING',
    'promoCode', case when p_promo_code is not null and trim(p_promo_code) != '' then upper(trim(p_promo_code)) else null end,
    'notes', p_notes,
    'createdAt', v_created_at
  );

  -- ----------------------------------------------------------------------------
  -- 9. COMMIT IDEMPOTENCY RECORD
  -- ----------------------------------------------------------------------------
  update public.checkout_idempotency_keys
  set
    status = 'COMMITTED',
    order_id = v_order_id,
    response_payload = v_response_payload,
    expires_at = pg_catalog.now() + interval '7 days'
  where key_hash = p_key_hash;

  return v_response_payload;
end;
$$;

-- 4. Create RPC public.storefront_cancel_pending_order to release inventory cleanly
create or replace function public.storefront_cancel_pending_order(
  p_order_number text,
  p_reason text default 'Checkout dismissed or payment failed'
)
returns boolean
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order record;
  v_item record;
  v_product_id uuid;
begin
  if p_order_number is null or length(trim(p_order_number)) = 0 then
    raise exception '%', jsonb_build_object(
      'code', 'VALIDATION_ERROR',
      'message', 'A valid order number is required.'
    )::text;
  end if;

  select *
  into v_order
  from public.orders
  where order_number = upper(trim(p_order_number))
  for update;

  if not found then
    raise exception '%', jsonb_build_object(
      'code', 'ORDER_NOT_FOUND',
      'message', 'Order ' || p_order_number || ' was not found.'
    )::text;
  end if;

  -- Idempotent return if already CANCELLED
  if v_order.status = 'CANCELLED' then
    return true;
  end if;

  if v_order.payment_status = 'PAID' then
    raise exception '%', jsonb_build_object(
      'code', 'CANNOT_CANCEL_PAID_ORDER',
      'message', 'Paid orders cannot be automatically cancelled via storefront dismissal.'
    )::text;
  end if;

  -- Restore stock for each item in the order
  for v_item in
    select oi.serial_number, sum(oi.quantity)::integer as quantity
    from public.order_items as oi
    where oi.order_id = v_order.id
    group by oi.serial_number
  loop
    select p.id into v_product_id
    from public.products as p
    where p.serial_number = v_item.serial_number
    for update;

    if found then
      update public.products
      set
        stock = stock + v_item.quantity,
        status = case
          when status = 'SOLD_OUT' then 'ACTIVE'
          else status
        end,
        updated_at = timezone('utc'::text, now())
      where id = v_product_id;
    end if;
  end loop;

  -- Mark order cancelled with note
  update public.orders
  set
    status = 'CANCELLED',
    notes = coalesce(notes, '') || ' | ' || coalesce(p_reason, 'Checkout dismissed'),
    updated_at = timezone('utc'::text, now())
  where id = v_order.id;

  -- Audit log
  insert into public.activity_audit_logs (
    actor_id,
    actor_role,
    actor_email,
    table_name,
    record_id,
    action,
    before_state,
    after_state,
    diff
  )
  values (
    coalesce(v_order.customer_id::text, 'guest'),
    'system',
    v_order.customer_email,
    'orders',
    v_order.id::text,
    'ORDER_STATUS_UPDATE',
    jsonb_build_object('status', v_order.status, 'payment_status', v_order.payment_status),
    jsonb_build_object('status', 'CANCELLED', 'payment_status', v_order.payment_status),
    jsonb_build_object('reason', p_reason, 'operation', 'ORDER_RESERVATION_CANCELLED')
  );

  return true;
end;
$$;

create or replace function public.register_for_event_atomic(
  p_event_id uuid,
  p_name text,
  p_email text,
  p_phone text,
  p_notes text default ''
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_event public.events%rowtype;
  v_reg public.event_registrations%rowtype;
  v_clean_email text;
  v_clean_name text;
  v_clean_phone text;
begin
  if p_event_id is null then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Event ID is required.')::text using errcode = '22023';
  end if;

  v_clean_email := lower(trim(coalesce(p_email, '')));
  v_clean_name := trim(coalesce(p_name, ''));
  v_clean_phone := trim(coalesce(p_phone, ''));

  if length(v_clean_name) < 2 or length(v_clean_name) > 100 then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Name must contain 2 to 100 characters.')::text using errcode = '22023';
  end if;
  if v_clean_email !~ '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$' then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Valid email address is required.')::text using errcode = '22023';
  end if;
  if length(v_clean_phone) < 7 or length(v_clean_phone) > 20 then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Valid phone number is required.')::text using errcode = '22023';
  end if;

  -- Acquire exclusive row lock on the event to prevent concurrent registration race conditions
  select * into v_event
  from public.events
  where id = p_event_id
  for update;

  if not found then
    raise exception '%', jsonb_build_object('code', 'EVENT_NOT_FOUND', 'message', 'Event was not found.')::text using errcode = 'P0002';
  end if;

  if not v_event.is_registration_open then
    raise exception '%', jsonb_build_object('code', 'EVENT_REGISTRATION_CLOSED', 'message', 'Registration for this event is closed.')::text using errcode = 'P0001';
  end if;

  if v_event.capacity is not null and v_event.registered_count >= v_event.capacity then
    raise exception '%', jsonb_build_object('code', 'EVENT_CAPACITY_REACHED', 'message', 'This event has reached full capacity.')::text using errcode = 'P0001';
  end if;

  -- Check duplicate registration
  if exists (
    select 1 from public.event_registrations
    where event_id = p_event_id and lower(email) = v_clean_email
  ) then
    raise exception '%', jsonb_build_object('code', 'DUPLICATE_REGISTRATION', 'message', 'A registration for this email has already been received for this event.')::text using errcode = '23505';
  end if;

  -- Insert registration
  insert into public.event_registrations (
    event_id, name, email, phone, notes
  ) values (
    p_event_id, v_clean_name, v_clean_email, v_clean_phone, coalesce(trim(p_notes), '')
  )
  returning * into v_reg;

  -- Increment registered_count atomically
  update public.events
  set registered_count = registered_count + 1,
      updated_at = timezone('utc'::text, now())
  where id = p_event_id;

  return jsonb_build_object(
    'id', v_reg.id,
    'eventId', v_reg.event_id,
    'eventTitle', v_event.title,
    'name', v_reg.name,
    'email', v_reg.email,
    'phone', v_reg.phone,
    'notes', nullif(v_reg.notes, ''),
    'createdAt', v_reg.created_at
  );
end;
$$;

-- ------------------------------------------------------------------------------
-- 4. Admin Refund Processing with Stock Restock
-- ------------------------------------------------------------------------------
create or replace function public.admin_process_order_refund(
  p_order_ref text,
  p_amount_paise bigint,
  p_reason text,
  p_routing_action text,
  p_actor_id text,
  p_actor_email text,
  p_actor_role text,
  p_ip_address text default null,
  p_user_agent text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order public.orders%rowtype;
  v_item record;
  v_refund public.refunds%rowtype;
  v_new_payment_status text;
  v_new_order_status text;
  v_clean_action text;
begin
  if coalesce(auth.role(), '') <> 'service_role' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Admin refund operation is not authorized.')::text using errcode = '42501';
  end if;
  if p_actor_role is null or p_actor_role <> 'owner' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Only owners may process refunds.')::text using errcode = '42501';
  end if;
  if p_amount_paise is null or p_amount_paise <= 0 then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Refund amount must be greater than zero.')::text using errcode = '22023';
  end if;

  v_clean_action := upper(trim(coalesce(p_routing_action, 'RESTOCK')));
  if v_clean_action not in ('RESTOCK', 'DRY_CLEAN', 'ALTERATION', 'DISCARD') then
    v_clean_action := 'RESTOCK';
  end if;

  select o.* into v_order
  from public.orders as o
  where o.order_number = upper(trim(p_order_ref)) or o.id::text = trim(p_order_ref)
  for update;

  if not found then
    raise exception '%', jsonb_build_object('code', 'ORDER_NOT_FOUND', 'message', 'Order was not found.')::text using errcode = 'P0002';
  end if;

  if v_order.payment_status not in ('PAID', 'PARTIALLY_REFUNDED') then
    raise exception '%', jsonb_build_object('code', 'ORDER_NOT_PAID', 'message', 'Cannot refund an unpaid order.')::text using errcode = 'P0001';
  end if;

  if p_amount_paise > v_order.total_paise then
    raise exception '%', jsonb_build_object('code', 'REFUND_EXCEEDS_TOTAL', 'message', 'Refund amount cannot exceed order total.')::text using errcode = '22023';
  end if;

  if p_amount_paise >= v_order.total_paise then
    v_new_payment_status := 'REFUNDED';
    v_new_order_status := 'REFUNDED';
  else
    v_new_payment_status := 'PARTIALLY_REFUNDED';
    v_new_order_status := v_order.status;
  end if;

  -- Insert into public.refunds
  insert into public.refunds (
    order_id, amount_paise, reason, status, approved_by, routing_action
  ) values (
    v_order.id, p_amount_paise, trim(coalesce(p_reason, 'Customer requested refund')), 'APPROVED', p_actor_email, v_clean_action
  )
  returning * into v_refund;

  -- Update order status
  update public.orders
  set payment_status = v_new_payment_status,
      status = v_new_order_status,
      updated_at = timezone('utc'::text, now())
  where id = v_order.id
  returning * into v_order;

  -- If restock action chosen, restore inventory
  if v_clean_action = 'RESTOCK' then
    for v_item in select product_id, quantity from public.order_items where order_id = v_order.id loop
      update public.products
      set stock = stock + v_item.quantity,
          status = case when status = 'SOLD_OUT' then 'ACTIVE' else status end,
          updated_at = timezone('utc'::text, now())
      where id = v_item.product_id;
    end loop;
  end if;

  -- Insert audit log
  insert into public.activity_audit_logs (
    actor_id, actor_email, actor_role, table_name, record_id, action,
    after_state, ip_address, user_agent
  ) values (
    p_actor_id, p_actor_email, p_actor_role, 'Refund', v_refund.id::text,
    'ORDER_REFUNDED',
    jsonb_build_object(
      'orderNumber', v_order.order_number,
      'refundAmountPaise', p_amount_paise,
      'routingAction', v_clean_action,
      'newPaymentStatus', v_new_payment_status
    ),
    coalesce(p_ip_address, '127.0.0.1'), left(coalesce(p_user_agent, 'CircleUp Admin'), 1024)
  );

  return jsonb_build_object(
    'refund', to_jsonb(v_refund),
    'order', to_jsonb(v_order)
  );
end;
$$;

revoke all on function public.admin_process_order_refund(text, bigint, text, text, text, text, text, text, text) from public, anon, authenticated;
grant execute on function public.admin_process_order_refund(text, bigint, text, text, text, text, text, text, text) to service_role;

-- ------------------------------------------------------------------------------
-- 5. Admin Fulfillment Creation
-- ------------------------------------------------------------------------------
create or replace function public.admin_create_order_fulfillment(
  p_order_ref text,
  p_tracking_number text,
  p_carrier text,
  p_notes text default null,
  p_actor_id text default null,
  p_actor_email text default null,
  p_actor_role text default null,
  p_ip_address text default null,
  p_user_agent text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order public.orders%rowtype;
  v_fulfillment public.fulfillments%rowtype;
begin
  if coalesce(auth.role(), '') <> 'service_role' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Admin fulfillment operation is not authorized.')::text using errcode = '42501';
  end if;

  select o.* into v_order
  from public.orders as o
  where o.order_number = upper(trim(p_order_ref)) or o.id::text = trim(p_order_ref)
  for update;

  if not found then
    raise exception '%', jsonb_build_object('code', 'ORDER_NOT_FOUND', 'message', 'Order was not found.')::text using errcode = 'P0002';
  end if;

  insert into public.fulfillments (
    order_id, tracking_number, carrier, status, shipped_at, notes
  ) values (
    v_order.id, trim(p_tracking_number), trim(p_carrier), 'SHIPPED', timezone('utc'::text, now()), nullif(trim(p_notes), '')
  )
  returning * into v_fulfillment;

  -- Advance order status to SHIPPED if it was CONFIRMED or PACKED
  if v_order.status in ('CONFIRMED', 'PACKED') then
    update public.orders
    set status = 'SHIPPED',
        updated_at = timezone('utc'::text, now())
    where id = v_order.id
    returning * into v_order;
  end if;

  -- Insert audit log
  insert into public.activity_audit_logs (
    actor_id, actor_email, actor_role, table_name, record_id, action,
    after_state, ip_address, user_agent
  ) values (
    coalesce(p_actor_id, 'system'), coalesce(p_actor_email, 'admin@circleuponline.com'), coalesce(p_actor_role, 'staff'),
    'Fulfillment', v_fulfillment.id::text, 'ORDER_FULFILLED',
    jsonb_build_object(
      'orderNumber', v_order.order_number,
      'trackingNumber', v_fulfillment.tracking_number,
      'carrier', v_fulfillment.carrier,
      'status', v_fulfillment.status
    ),
    coalesce(p_ip_address, '127.0.0.1'), left(coalesce(p_user_agent, 'CircleUp Admin'), 1024)
  );

  return jsonb_build_object(
    'fulfillment', to_jsonb(v_fulfillment),
    'order', to_jsonb(v_order)
  );
end;
$$;

revoke all on function public.admin_create_order_fulfillment(text, text, text, text, text, text, text, text, text) from public, anon, authenticated;
grant execute on function public.admin_create_order_fulfillment(text, text, text, text, text, text, text, text, text) to service_role;

  on public.newsletter_subscriptions for insert
