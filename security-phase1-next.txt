/**
 * Constructs and configures the modular Express application without binding a port,
 * enabling clean Supertest integration testing.
 */
export function createApp(): Express {
  const app = express();

  app.set('trust proxy', 1);
  app.set('etag', 'strong');

  // 1. Request correlation & structured logging
  app.use(attachRequestId);
  app.use(httpLogger);

  // 2. Security headers, gzip/brotli compression, strict CORS, body/cookie parsing, NoSQL sanitization & rate limit
  app.use(helmet());
  app.use(compression());
  app.use(cors(corsOptions));
  app.use(express.json({
    limit: '100kb',
    verify: (req, _res, buf) => {
      (req as { rawBody?: Buffer }).rawBody = buf;
    },
  }));
  app.use(express.urlencoded({ extended: false, limit: '100kb' }));
  app.use(cookieParser());
  app.use(sanitizeRequest);
  app.use('/api', globalRateLimiter);

  // 3. Mount public endpoints on both /api and /api/v1 for seamless frontend client compatibility
  const apiRouters = [
    { path: '/homepage', router: homepageRouter },
    { path: '/search', router: searchRouter },
    { path: '/coupons', router: couponRouter },
    { path: '/', router: categoryRouter }, // /categories and /collections
    { path: '/rentals', router: rentalRouter },
    { path: '/events', router: eventRouter },
    { path: '/reviews', router: reviewRouter },
    { path: '/community', router: communityRouter },
    { path: '/cron', router: cronRouter },
    { path: '/products', router: productRouter },
    { path: '/orders', router: orderRouter },
    { path: '/payments/razorpay', router: paymentRouter },
    { path: '/contact', router: contactRouter },
    { path: '/marketing', router: marketingRouter },
    { path: '/shipping-tax', router: shippingTaxRouter },
    { path: '/health', router: healthRouter },
  ];

  for (const prefix of ['/api', '/api/v1']) {
    app.use(`${prefix}/admin`, adminRouter);
    for (const { path: routePath, router } of apiRouters) {
      if (routePath === '/') {
        app.use(prefix, router);
      } else {
        app.use(`${prefix}${routePath}`, router);
      }
    }
  }

  // 4. 404 fallback & global error serialization
  app.use((_req, _res, next) => {
    next(new NotFoundError('API endpoint not found', 'ROUTE_NOT_FOUND'));
  });
  app.use(errorHandler);

  return app;
}

import { createHmac } from 'node:crypto';
import { describe, expect, it } from 'vitest';
import request from 'supertest';
import { createApp } from '../../src/app.js';
import { envSchema } from '../../src/config/env.js';
import {
  createRazorpayOrder,
  verifyRazorpayPaymentSignature,
  verifyRazorpayWebhookSignature,
} from '../../src/services/razorpay.service.js';

describe('Razorpay Direct Checkout & Verification Lifecycle', () => {
  const app = createApp();

  it('rejects new Cash on Delivery orders at checkout with PAYMENT_METHOD_DISALLOWED', async () => {
    const res = await request(app)
      .post('/api/v1/orders')
      .set('Idempotency-Key', 'test-cod-disallowed-key-001')
      .send({
        customer: {
          name: 'Priya Sharma',
          email: 'priya@example.com',
          phone: '+919876543210',
        },
        shippingAddress: {
          addressLine: '10 Marine Drive',
          city: 'Mumbai',
          state: 'Maharashtra',
          postalCode: '400020',
        },
        items: [
          {
            serialNumber: 'CU-001',
            quantity: 1,
            size: 'M',
          },
        ],
        paymentMethod: 'CASH_ON_DELIVERY',
      });

    expect(res.status).toBe(400);
    expect(res.body.success).toBe(false);
    expect(res.body.error.code).toBe('PAYMENT_METHOD_DISALLOWED');
    expect(res.body.error.message).toMatch(/Cash on Delivery is no longer available/i);
  });

  it('preserves integer paise in Razorpay order creation', async () => {
    // 2550 INR = 255000 paise
    const amountInPaise = 255000;
    const internalOrderNumber = 'ORD-261123-A1B2C3D4E5F6';

    const order = await createRazorpayOrder({
      amountPaise: amountInPaise,
      internalOrderNumber,
    });

    expect(order).toBeDefined();
    expect(order.amount).toBe(255000);
    expect(order.currency).toBe('INR');
  });

  it('accepts valid payment signatures and rejects fraudulent signatures', () => {
    const secret = 'synthetic-test-secret-fixture-only';
    const orderId = 'order_test_987654';
    const paymentId = 'pay_test_123456';
    const validSignature = createHmac('sha256', secret)
      .update(`${orderId}|${paymentId}`)
      .digest('hex');

    expect(verifyRazorpayPaymentSignature({
      orderId,
      paymentId,
      signature: validSignature,
    })).toBe(true);

    expect(verifyRazorpayPaymentSignature({
      orderId: 'order_test_tampered',
      paymentId,
      signature: validSignature,
    })).toBe(false);

    expect(verifyRazorpayPaymentSignature({
      orderId,
      paymentId,
      signature: 'bad-signature-of-invalid-length',
    })).toBe(false);
  });

  it('validates webhook signatures against raw request buffer', () => {
    const webhookSecret = 'synthetic-webhook-secret-fixture-only';
    const rawBody = Buffer.from(JSON.stringify({ event: 'payment.captured', payload: {} }));
    const validSignature = createHmac('sha256', webhookSecret)
      .update(rawBody)
      .digest('hex');

    expect(verifyRazorpayWebhookSignature(rawBody, validSignature)).toBe(true);
    expect(verifyRazorpayWebhookSignature(Buffer.from('tampered body'), validSignature)).toBe(false);
  });

  it('enforces RAZORPAY_KEY_ID and RAZORPAY_KEY_SECRET in production env configuration', () => {
    const baseProdEnv = {
      NODE_ENV: 'production',
      PORT: '5000',
      SUPABASE_URL: 'https://test-project.supabase.co',
      SUPABASE_SERVICE_ROLE_KEY: 'test-service-role-key-with-sufficient-length',
      SESSION_SECRET: 'circleup-production-session-secret-at-least-32-chars',
      JWT_ACCESS_SECRET: 'circleup-production-jwt-access-secret-32-chars',
      JWT_ADMIN_ACCESS_SECRET: 'circleup-production-jwt-admin-secret-at-least-32',
      GUEST_ORDER_TOKEN_SECRET: 'circleup-production-guest-token-secret-at-least-32',
      CRON_SECRET: 'circleup-cron-secret-16',
      MONETARY_RECONCILIATION_VERIFIED: 'true',
    };

    // Missing Razorpay keys
    const missingKeys = envSchema.safeParse(baseProdEnv);
    expect(missingKeys.success).toBe(false);
    if (!missingKeys.success) {
      const fieldErrors = missingKeys.error.flatten().fieldErrors;
      expect(fieldErrors.RAZORPAY_KEY_ID).toBeDefined();
      expect(fieldErrors.RAZORPAY_KEY_SECRET).toBeDefined();
    }

    // Insecure placeholder keys
    const placeholderKeys = envSchema.safeParse({
      ...baseProdEnv,
      RAZORPAY_KEY_ID: 'replace-with-your-key',
      RAZORPAY_KEY_SECRET: 'replace-with-your-secret',
    });
    expect(placeholderKeys.success).toBe(false);
  });

  it('POST /api/v1/payments/razorpay/verify validates required fields', async () => {
    const res = await request(app)
      .post('/api/v1/payments/razorpay/verify')
      .send({});

    expect(res.status).toBe(400);
    expect(res.body.success).toBe(false);
    expect(res.body.error.code).toBe('VALIDATION_ERROR');
  });

  it('POST /api/v1/payments/razorpay/cancel requires order access authorization', async () => {
    const res = await request(app)
      .post('/api/v1/payments/razorpay/cancel')
      .send({
        orderNumber: 'ORD-NONEXISTENT-9999',
      });

    // Should return 404 for unknown order or 401 for unauthorized
    expect([401, 404]).toContain(res.status);
    expect(res.body.success).toBe(false);
  });
});

      expect(checkoutState.products.get('CU-002')?.stock).toBe(9);

    } finally {
      checkoutState.setFailure(null);
    }
  });

    const ordersBefore = checkoutState.orders.size;

    const second = await checkoutRequest().set('Idempotency-Key', key).send(payload);
    expect(second.status).toBe(201);

    const [first, second] = await Promise.all([submit(), submit()]);

    expect(second.status).toBe(201);
    expect(first.body.data.orderNumber).toBe(second.body.data.orderNumber);

    expect(checkoutState.products.get('CU-001')?.stock).toBe(4);
    const retry = await submit();
    expect(retry.status).toBe(201);
    expect(checkoutState.orders.size).toBe(before + 1);
  });

  it('fails checkout when Supabase is unavailable without falling back to legacy databases', async () => {
    checkoutState.setEnabled(false);
    try {
      const res = await checkoutRequest().set('Idempotency-Key', 'offline-checkout-key-0004').send({

      expect(res.status).toBe(503);
      expect(res.body.error.code).toBe('DATABASE_UNAVAILABLE');
      expect(res.body.data).toBeUndefined();
    } finally { checkoutState.setEnabled(true); }
  });

as $$

  insert into public.profiles (id, full_name, email, avatar_url)

    new.id,

-- 4. Function & Trigger: Auto-derive category_code from serial_number if null
create or replace function public.derive_product_category_code()
returns trigger
language plpgsql
as $$
declare
  cleaned_serial text;
begin
  if new.category_code is null or new.category_code = '' then
    cleaned_serial := upper(trim(new.serial_number));

    -- Check multi-part codes first
    if cleaned_serial like 'MB-J-%' or cleaned_serial like 'MB-J_%' then new.category_code := 'MB-J';
    elsif cleaned_serial like 'WB-J-%' or cleaned_serial like 'WB-J_%' then new.category_code := 'WB-J';
    elsif cleaned_serial like 'MB-O-%' or cleaned_serial like 'MB-O_%' then new.category_code := 'MB-O';
    elsif cleaned_serial like 'MB-L-%' or cleaned_serial like 'MB-L_%' then new.category_code := 'MB-L';
    elsif cleaned_serial like 'WB-L-%' or cleaned_serial like 'WB-L_%' then new.category_code := 'WB-L';
    elsif cleaned_serial like 'WB-O-%' or cleaned_serial like 'WB-O_%' then new.category_code := 'WB-O';
    elsif cleaned_serial like 'CWZ-%' or cleaned_serial like 'CWZ_%' then new.category_code := 'CWZ';
    elsif cleaned_serial like 'OWJ-%' or cleaned_serial like 'OWJ_%' then new.category_code := 'OWJ';
    elsif cleaned_serial like 'TCE-%' or cleaned_serial like 'TCE_%' then new.category_code := 'TCE';
    elsif cleaned_serial like 'DIY-%' or cleaned_serial like 'DIY_%' then new.category_code := 'DIY';
    elsif cleaned_serial like 'SET-%' or cleaned_serial like 'SET_%' then new.category_code := 'SET';
    elsif cleaned_serial like 'DJ-%' or cleaned_serial like 'DJ_%' then new.category_code := 'DJ';
    elsif cleaned_serial like 'LJ-%' or cleaned_serial like 'LJ_%' then new.category_code := 'LJ';
    elsif cleaned_serial like 'SZ-%' or cleaned_serial like 'SZ_%' then new.category_code := 'SZ';
    elsif cleaned_serial like 'SC-%' or cleaned_serial like 'SC_%' then new.category_code := 'SC';
    elsif cleaned_serial like 'FS-%' or cleaned_serial like 'FS_%' then new.category_code := 'FS';
    elsif cleaned_serial like 'DB-%' or cleaned_serial like 'DB_%' then new.category_code := 'DB';
    elsif cleaned_serial like 'SS-%' or cleaned_serial like 'SS_%' then new.category_code := 'SS';
    elsif cleaned_serial like 'SH-%' or cleaned_serial like 'SH_%' then new.category_code := 'SH';
    elsif cleaned_serial like 'BB-%' or cleaned_serial like 'BB_%' then new.category_code := 'BB';
    elsif cleaned_serial like 'JW-%' or cleaned_serial like 'JW_%' then new.category_code := 'JW';
    elsif cleaned_serial like 'CJ-%' or cleaned_serial like 'CJ_%' then new.category_code := 'CJ';
    elsif cleaned_serial like 'SB-%' or cleaned_serial like 'SB_%' then new.category_code := 'SB';
    elsif cleaned_serial like 'MS-%' or cleaned_serial like 'MS_%' then new.category_code := 'MS';
    elsif cleaned_serial like 'EX-%' or cleaned_serial like 'EX_%' then new.category_code := 'EX';
    elsif cleaned_serial like 'H-%' or cleaned_serial like 'H_%' then new.category_code := 'H';
    elsif cleaned_serial like 'T-%' or cleaned_serial like 'T_%' then new.category_code := 'T';
    elsif cleaned_serial like 'B-%' or cleaned_serial like 'B_%' then new.category_code := 'B';
    elsif cleaned_serial like 'S-%' or cleaned_serial like 'S_%' then new.category_code := 'S';
    elsif cleaned_serial like 'J-%' or cleaned_serial like 'J_%' then new.category_code := 'J';
    elsif cleaned_serial like 'R-%' or cleaned_serial like 'R_%' then new.category_code := 'R';
    end if;
  end if;

  return new;
end;
$$;

-- 10. ATOMIC AUDIT LOGGING TRIGGER FUNCTION
-- ------------------------------------------------------------------------------
create or replace function public.fn_activity_audit_log()
returns trigger
language plpgsql
security definer
as $$
declare
  rec_id text;
  old_data jsonb;
  new_data jsonb;
  act_type text;
begin
  if tg_op = 'DELETE' then
    rec_id := coalesce(old.id::text, '');
    old_data := to_jsonb(old);
    new_data := null;
    act_type := 'DELETE';
  elsif tg_op = 'INSERT' then
    rec_id := coalesce(new.id::text, '');
    old_data := null;
    new_data := to_jsonb(new);
    act_type := 'INSERT';
  else
    rec_id := coalesce(new.id::text, '');
    old_data := to_jsonb(old);
    new_data := to_jsonb(new);
    if old_data->>'status' is distinct from new_data->>'status' then
      act_type := 'STATUS_CHANGE';
    else
      act_type := 'UPDATE';
    end if;
  end if;

  insert into public.activity_audit_logs (
    actor_id,
    actor_role,
    actor_email,
    table_name,
    record_id,
    action,
    before_state,
    after_state
  )
  values (
    coalesce(auth.uid()::text, 'service_role'),
    coalesce(auth.jwt() -> 'app_metadata' ->> 'role', 'admin'),
    auth.jwt() ->> 'email',
    tg_table_name,
    rec_id,
    act_type,
    old_data,
    new_data
  );

  return coalesce(new, old);
end;
$$;

-- ==============================================================================
-- CircleUp Supabase Migration: Atomic Storefront Checkout RPC
-- Milestone 1, Step 2: public.process_storefront_checkout (Hardened & Revised)
-- ==============================================================================

-- ------------------------------------------------------------------------------
-- 0. SCHEMA PREREQUISITES
-- Ensure public.site_settings has is_whatsapp_verified column to enforce explicit
-- store-verified configuration before WhatsApp / UPI checkout can be enabled.
-- ------------------------------------------------------------------------------
alter table public.site_settings
  add column if not exists is_whatsapp_verified boolean not null default false;

-- ------------------------------------------------------------------------------
-- Function: public.process_storefront_checkout
-- Executes the entire checkout flow in a single atomic PostgreSQL transaction:
--  1. Validates and claims the idempotency key with request fingerprint.
--  2. Locks product rows (SELECT ... FOR UPDATE) in deterministic sort order to prevent deadlocks.
--  3. Validates product status ('AVAILABLE', 'ACTIVE', 'PUBLISHED'), non-negative price, and stock.
--  4. Validates requested size/variant against product's actual available sizes; rejects invalid variants.
--  5. Validates WhatsApp configuration against explicit store-verified state and placeholder contacts.
--  6. Validates and locks coupon row; calculates discount using numeric precision (PERCENTAGE/FLAT);
--     rejects unsupported coupon types (FREE_SHIPPING, BOGO) before usage increments.
--  7. Calculates delivery fee and order total using canonical paise (1 INR = 100 paise).
--  8. Decrements product stock exactly once; updates status to 'SOLD_OUT' if stock reaches 0.
--  9. Inserts public.orders with collision-resistant order number (ORD-YYMMDD-XXXXXXXX) and retry loop.
-- 10. Inserts public.order_items with verified size, image, and slug.
-- 11. Updates idempotency reservation with sanitized response payload (24h TTL).
-- 12. Guarantees ZERO plaintext guest access tokens in database or cached response_payload.
-- ------------------------------------------------------------------------------
create or replace function public.process_storefront_checkout(
  p_key_hash text,
  p_fingerprint text,
  p_customer_id uuid,
  p_customer jsonb,
  p_shipping_address jsonb,
  p_items jsonb,
  p_payment_method text,
  p_promo_code text default null,
  p_notes text default null,
  p_guest_access_token_hash text default null,
  p_guest_access_expires_at timestamptz default null,
  p_order_number text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_existing_idem record;
  v_conflicting_idem record;
  v_claimed_idem record;
  v_order_number text;
  v_order_id uuid;
  v_order_inserted boolean := false;
  v_created_at timestamptz;
  v_whatsapp_number text;
  v_cleaned_whatsapp text;
  v_is_whatsapp_verified boolean;
  v_constraint_name text;
  v_attempt integer;
  v_item record;
  v_serial text;
  v_qty integer;
  v_item_size text;
  v_product record;
  v_missing_serial text;
  v_subtotal bigint := 0;
  v_discount bigint := 0;
  v_delivery_fee bigint := 0;
  v_total_amount bigint := 0;
  v_discounted_subtotal bigint := 0;
  v_coupon record;
  v_new_stock integer;
  v_new_status text;
  v_items_json jsonb := '[]'::jsonb;
  v_response_payload jsonb;
  v_loop_count integer := 0;
  v_retry_insert boolean := true;
  v_serials text[] := array[]::text[];
  v_quantities jsonb := '{}'::jsonb;
  v_variant_match boolean;
  v_var record;
begin
  -- ----------------------------------------------------------------------------
  -- 1. INPUT SANITIZATION & IDEMPOTENCY KEY CHECK
  -- ----------------------------------------------------------------------------
  if p_key_hash is null or length(p_key_hash) != 64 then
    raise exception '%', jsonb_build_object(
      'code', 'IDEMPOTENCY_KEY_REQUIRED',
      'message', 'A valid 64-character SHA-256 idempotency key hash is required.'
    )::text;
  end if;

  if p_fingerprint is null or length(p_fingerprint) != 64 then
    raise exception '%', jsonb_build_object(
      'code', 'VALIDATION_ERROR',
      'message', 'A valid 64-character SHA-256 request fingerprint is required.'
    )::text;
  end if;

  -- Bounded retry loop for idempotency claim (handles races with background cleanup)
  while v_retry_insert and v_loop_count < 3 loop
    v_loop_count := v_loop_count + 1;
    v_retry_insert := false;

    select * into v_existing_idem
    from public.checkout_idempotency
    where key_hash = p_key_hash;

    if found then
      if v_existing_idem.expires_at > pg_catalog.now() then
        if v_existing_idem.fingerprint != p_fingerprint then
          raise exception '%', jsonb_build_object(
            'code', 'IDEMPOTENCY_KEY_REUSED',
            'message', 'This idempotency key was already used for a different checkout.'
          )::text;
        end if;

        if v_existing_idem.response_payload is not null then
          return v_existing_idem.response_payload;
        else
          raise exception '%', jsonb_build_object(
            'code', 'CHECKOUT_IN_PROGRESS',
            'message', 'Checkout with this key is already being processed. Retry shortly.'
          )::text;
        end if;
      else
        delete from public.checkout_idempotency
        where key_hash = p_key_hash
          and expires_at <= pg_catalog.now();
      end if;
    end if;

    -- Collision-resistant order number: ORD-YYMMDD-XXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXX (122 bits of CSPRNG entropy)
    -- Uses full unhyphenated pg_catalog.gen_random_uuid() to guarantee 100% resolution under empty search_path
    -- without external extension dependencies (pgcrypto or uuid-ossp).
    v_order_number := coalesce(
      nullif(trim(p_order_number), ''),
      'ORD-' ||
        pg_catalog.to_char(pg_catalog.now() at time zone 'Asia/Kolkata', 'YYMMDD') || '-' ||
        pg_catalog.upper(pg_catalog.replace(pg_catalog.gen_random_uuid()::text, '-', ''))
    );

    insert into public.checkout_idempotency (
      key_hash,
      fingerprint,
      order_number,
      response_payload,
      expires_at
    )
    values (
      p_key_hash,
      p_fingerprint,
      v_order_number,
      null,
      pg_catalog.now() + interval '5 minutes'
    )
    on conflict (key_hash) do nothing
    returning * into v_claimed_idem;

    if v_claimed_idem.id is null then
      select * into v_conflicting_idem
      from public.checkout_idempotency
      where key_hash = p_key_hash
      for update;

      if not found then
        v_retry_insert := true;
      else
        if v_conflicting_idem.expires_at > pg_catalog.now() then
          if v_conflicting_idem.fingerprint != p_fingerprint then
            raise exception '%', jsonb_build_object(
              'code', 'IDEMPOTENCY_KEY_REUSED',
              'message', 'This idempotency key was already used for a different checkout.'
            )::text;
          elsif v_conflicting_idem.response_payload is not null then
            return v_conflicting_idem.response_payload;
          else
            raise exception '%', jsonb_build_object(
              'code', 'CHECKOUT_IN_PROGRESS',
              'message', 'Checkout with this key is already being processed. Retry shortly.'
            )::text;
          end if;
        else
          delete from public.checkout_idempotency
          where key_hash = p_key_hash;
          v_retry_insert := true;
        end if;
      end if;
    end if;
  end loop;

  if v_claimed_idem.id is null and v_response_payload is null then
    raise exception '%', jsonb_build_object(
      'code', 'CHECKOUT_IN_PROGRESS',
      'message', 'Checkout reservation could not be acquired. Please retry.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 2. VALIDATE PAYMENT METHOD & WHATSAPP CONFIGURATION
  -- ----------------------------------------------------------------------------
  if p_payment_method = 'ONLINE_UPI' then
    raise exception '%', jsonb_build_object(
      'code', 'PAYMENT_PROVIDER_UNAVAILABLE',
      'message', 'Online payment is temporarily unavailable. Please try again later.'
    )::text;
  elsif p_payment_method not in ('CASH_ON_DELIVERY', 'WHATSAPP_UPI') then
    raise exception '%', jsonb_build_object(
      'code', 'VALIDATION_ERROR',
      'message', 'Invalid payment method: must be CASH_ON_DELIVERY or WHATSAPP_UPI.'
    )::text;
  end if;

  if p_payment_method = 'WHATSAPP_UPI' then
    select whatsapp_number, is_whatsapp_verified
    into v_whatsapp_number, v_is_whatsapp_verified
    from public.site_settings
    limit 1;

    v_cleaned_whatsapp := pg_catalog.regexp_replace(coalesce(v_whatsapp_number, ''), '\D', '', 'g');

    -- WhatsApp checkout requires an explicit store-verified configuration.
    -- Syntactic validity or absence of blacklisted numbers is NOT sufficient proof of store ownership.
    if coalesce(v_is_whatsapp_verified, false) is not true or
       v_cleaned_whatsapp is null or
       not (v_cleaned_whatsapp ~ '^[1-9][0-9]{9,14}$') or
       v_cleaned_whatsapp in ('919876543210', '910000000000', '1234567890', '9876543210') or
       v_cleaned_whatsapp like '%000000%' then
      raise exception '%', jsonb_build_object(
        'code', 'PAYMENT_METHOD_UNAVAILABLE',
        'message', 'WhatsApp / UPI payment is temporarily unavailable because store contact details have not been verified.'
      )::text;
    end if;
  end if;

  -- ----------------------------------------------------------------------------
  -- 3. VALIDATE CUSTOMER & SHIPPING ADDRESS
  -- ----------------------------------------------------------------------------
  if p_customer is null or
     trim(coalesce(p_customer->>'name', '')) = '' or
     trim(coalesce(p_customer->>'email', '')) = '' or
     trim(coalesce(p_customer->>'phone', '')) = '' then
    raise exception '%', jsonb_build_object(
      'code', 'MISSING_CUSTOMER_INFO',
      'message', 'Customer name, email, and phone are required.'
    )::text;
  end if;

  if p_shipping_address is null or
     trim(coalesce(p_shipping_address->>'addressLine', '')) = '' or
     trim(coalesce(p_shipping_address->>'city', '')) = '' or
     trim(coalesce(p_shipping_address->>'state', '')) = '' or
     trim(coalesce(p_shipping_address->>'postalCode', '')) = '' then
    raise exception '%', jsonb_build_object(
      'code', 'INCOMPLETE_ADDRESS',
      'message', 'Complete delivery address (line, city, state, postal code) is required.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 4. PARSE & AGGREGATE ITEMS
  -- ----------------------------------------------------------------------------
  if p_items is null or jsonb_array_length(p_items) = 0 then
    raise exception '%', jsonb_build_object(
      'code', 'EMPTY_ORDER_ITEMS',
      'message', 'Order must contain at least one item.'
    )::text;
  end if;

  for v_item in select * from jsonb_to_recordset(p_items) as x("serialNumber" text, "quantity" integer, "size" text) loop
    v_serial := upper(trim(coalesce(v_item."serialNumber", '')));
    v_qty := coalesce(v_item."quantity", 0);

    if v_serial = '' then
      raise exception '%', jsonb_build_object(
        'code', 'VALIDATION_ERROR',
        'message', 'Item serial number is required.'
      )::text;
    end if;

    if v_qty <= 0 or v_qty > 10 then
      raise exception '%', jsonb_build_object(
        'code', 'INVALID_QUANTITY',
        'message', 'Quantity must be between 1 and 10 per item.'
      )::text;
    end if;

    if not (v_serials @> array[v_serial]) then
      v_serials := array_append(v_serials, v_serial);
      v_quantities := v_quantities || jsonb_build_object(v_serial, v_qty);
    else
      v_qty := (v_quantities->>v_serial)::integer + v_qty;
      if v_qty > 10 then
        raise exception '%', jsonb_build_object(
          'code', 'INVALID_QUANTITY',
          'message', 'Maximum quantity per product is 10.'
        )::text;
      end if;
      v_quantities := v_quantities || jsonb_build_object(v_serial, v_qty);
    end if;
  end loop;

  -- ----------------------------------------------------------------------------
  -- 5. LOCK PRODUCT ROWS IN DETERMINISTIC SORT ORDER & VALIDATE VARIANTS
  -- ----------------------------------------------------------------------------
  for v_product in
    select p.*
    from public.products p
    where upper(trim(p.serial_number)) = any(v_serials)
    order by p.serial_number asc
    for update
  loop
    v_serial := upper(trim(v_product.serial_number));
    v_qty := (v_quantities->>v_serial)::integer;

    -- Stock exhaustion check takes priority so items decremented to 0 report OUT_OF_STOCK
    if v_product.stock is null or v_product.stock < v_qty then
      raise exception '%', jsonb_build_object(
        'code', 'OUT_OF_STOCK',
        'message', 'Insufficient stock for "' || v_product.serial_number || '".'
      )::text;
    end if;

    if v_product.status not in ('AVAILABLE', 'ACTIVE', 'PUBLISHED') then
      raise exception '%', jsonb_build_object(
        'code', 'PRODUCT_NOT_ORDERABLE',
        'message', 'Product "' || v_product.serial_number || '" is not available to order.'
      )::text;
    end if;

    if v_product.price is null or v_product.price < 0 or v_product.price > 1000000000 then
      raise exception '%', jsonb_build_object(
        'code', 'INVALID_PRODUCT_PRICE',
        'message', 'Product "' || v_product.serial_number || '" has an invalid stored price.'
      )::text;
    end if;

    v_subtotal := v_subtotal + (v_product.price * v_qty);
  end loop;

  if (
    select count(*)
    from public.products p
    where upper(trim(p.serial_number)) = any(v_serials)
  ) < array_length(v_serials, 1) then
    select s into v_missing_serial
    from unnest(v_serials) s
    where not exists (
      select 1 from public.products p where upper(trim(p.serial_number)) = s
    )
    limit 1;

    raise exception '%', jsonb_build_object(
      'code', 'PRODUCT_NOT_ORDERABLE',
      'message', 'Product "' || coalesce(v_missing_serial, 'UNKNOWN') || '" is not available to order.'
    )::text;
  end if;

  -- Validate sizes/variants for each requested item
  for v_item in select * from jsonb_to_recordset(p_items) as x("serialNumber" text, "quantity" integer, "size" text) loop
    v_serial := upper(trim(v_item."serialNumber"));
    v_item_size := trim(coalesce(v_item."size", ''));

    select * into v_product
    from public.products
    where upper(trim(serial_number)) = v_serial;

    -- If product defines available sizes:
    if array_length(v_product.sizes, 1) > 0 then
      if v_item_size = '' then
        -- If single size option exists, default to it
        if array_length(v_product.sizes, 1) = 1 then
          v_item_size := v_product.sizes[1];
        else
          raise exception '%', jsonb_build_object(
            'code', 'INVALID_VARIANT',
            'message', 'Size is required for product "' || v_product.serial_number || '". Available: ' || array_to_string(v_product.sizes, ', ') || '.'
          )::text;
        end if;
      elsif not (v_product.sizes @> array[v_item_size]) then
        -- Check if size exists in variants jsonb
        v_variant_match := false;
        if jsonb_array_length(v_product.variants) > 0 then
          for v_var in select * from jsonb_to_recordset(v_product.variants) as z("size" text, "label" text) loop
            if upper(trim(coalesce(v_var."size", ''))) = upper(v_item_size) or upper(trim(coalesce(v_var."label", ''))) = upper(v_item_size) then
              v_variant_match := true;
              exit;
            end if;
          end loop;
        end if;

        if not v_variant_match then
          raise exception '%', jsonb_build_object(
            'code', 'INVALID_VARIANT',
            'message', 'Size "' || v_item_size || '" is not available for product "' || v_product.serial_number || '". Available sizes: ' || array_to_string(v_product.sizes, ', ') || '.'
          )::text;
        end if;
      end if;
    elsif jsonb_array_length(v_product.variants) > 0 then
      -- Product uses variants jsonb exclusively
      v_variant_match := false;
      for v_var in select * from jsonb_to_recordset(v_product.variants) as z("size" text, "label" text) loop
        if v_item_size != '' and (upper(trim(coalesce(v_var."size", ''))) = upper(v_item_size) or upper(trim(coalesce(v_var."label", ''))) = upper(v_item_size)) then
          v_variant_match := true;
          exit;
        end if;
      end loop;

      if v_item_size != '' and not v_variant_match then
        raise exception '%', jsonb_build_object(
          'code', 'INVALID_VARIANT',
          'message', 'Variant "' || v_item_size || '" is not available for product "' || v_product.serial_number || '".'
        )::text;
      end if;
    end if;
  end loop;

  if v_subtotal < 0 or v_subtotal > 10000000000 then
    raise exception '%', jsonb_build_object(
      'code', 'INVALID_ORDER_TOTAL',
      'message', 'Order subtotal exceeds supported currency range.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 6. VALIDATE & LOCK COUPON (REJECT UNSUPPORTED TYPES EXPLICITLY)
  -- ----------------------------------------------------------------------------
  if p_promo_code is not null and trim(p_promo_code) != '' then
    select * into v_coupon
    from public.coupons
    where upper(trim(code)) = upper(trim(p_promo_code))
    for update;

    if not found or not v_coupon.is_active then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'Invalid or expired promotional code.'
      )::text;
    end if;

    -- Reject unsupported coupon types BEFORE updating usage count
    if v_coupon.discount_type not in ('PERCENTAGE', 'FLAT') then
      raise exception '%', jsonb_build_object(
        'code', 'UNSUPPORTED_COUPON_TYPE',
        'message', 'Promo code discount type "' || v_coupon.discount_type || '" is not supported at checkout.'
      )::text;
    end if;

    if v_coupon.starts_at is not null and v_coupon.starts_at > pg_catalog.now() then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'This promo code is not yet active.'
      )::text;
    end if;

    if v_coupon.ends_at is not null and v_coupon.ends_at < pg_catalog.now() then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'This promo code has expired.'
      )::text;
    end if;

    if v_coupon.usage_limit is not null and v_coupon.usage_count >= v_coupon.usage_limit then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'This promo code has reached its maximum usage limit.'
      )::text;
    end if;

    if v_subtotal < v_coupon.min_order_paise then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'Code ' || v_coupon.code || ' requires a minimum bag value of ₹' ||
          (round(v_coupon.min_order_paise / 100))::text || '.'
      )::text;
    end if;

    -- Accurate numeric percentage calculation avoiding integer division truncation
    if v_coupon.discount_type = 'PERCENTAGE' then
      v_discount := round((v_subtotal::numeric * v_coupon.discount_value::numeric) / 100.0)::bigint;
      if v_coupon.max_discount_paise is not null and v_discount > v_coupon.max_discount_paise then
        v_discount := v_coupon.max_discount_paise;
      end if;
      v_discount := least(v_discount, v_subtotal);
    elsif v_coupon.discount_type = 'FLAT' then
      v_discount := least(v_coupon.discount_value, v_subtotal);
    else
      v_discount := 0;
    end if;

    -- Increment coupon usage atomically
    update public.coupons
    set usage_count = usage_count + 1,
        updated_at = pg_catalog.now()
    where id = v_coupon.id;
  end if;

  -- ----------------------------------------------------------------------------
  -- 7. DELIVERY FEE & TOTAL CALCULATION (CANONICAL PAISE: 1 INR = 100 PAISE)
  -- ----------------------------------------------------------------------------
  v_discounted_subtotal := v_subtotal - v_discount;
  if v_discounted_subtotal >= 150000 then
    v_delivery_fee := 0; -- Free shipping threshold on bag value >= ₹1,500
  else
    v_delivery_fee := 9000; -- Standard ₹90 delivery fee
  end if;

  v_total_amount := v_discounted_subtotal + v_delivery_fee;

  if v_total_amount < 0 or v_total_amount > 10000000000 then
    raise exception '%', jsonb_build_object(
      'code', 'INVALID_ORDER_TOTAL',
      'message', 'Order total exceeds supported currency range.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 8. DECREMENT INVENTORY EXACTLY ONCE
  -- ----------------------------------------------------------------------------
  for v_product in
    select p.*
    from public.products p
    where upper(trim(p.serial_number)) = any(v_serials)
    order by p.serial_number asc
  loop
    v_serial := upper(trim(v_product.serial_number));
    v_qty := (v_quantities->>v_serial)::integer;
    v_new_stock := v_product.stock - v_qty;
    v_new_status := case when v_new_stock = 0 then 'SOLD_OUT' else v_product.status end;

    update public.products
    set stock = v_new_stock,
        status = v_new_status,
        updated_at = pg_catalog.now()
    where id = v_product.id;
  end loop;

  -- ----------------------------------------------------------------------------
  -- 9. INSERT PUBLIC.ORDERS RECORD (WITH RETRY ON UNIQUE ORDER NUMBER COLLISION)
  -- ----------------------------------------------------------------------------
  for v_attempt in 1..5 loop
    begin
      insert into public.orders (
        order_number,
        customer_id,
        customer_name,
        customer_email,
        customer_phone,
        total_amount,
        subtotal,
        discount_amount,
        shipping_amount,
        status,
        payment_status,
        payment_method,
        shipping_address,
        promo_code,
        notes,
        guest_access_token_hash,
        guest_access_expires_at,
        created_at,
        updated_at
      )
      values (
        v_order_number,
        p_customer_id,
        trim(p_customer->>'name'),
        lower(trim(p_customer->>'email')),
        trim(p_customer->>'phone'),
        v_total_amount,
        v_subtotal,
        v_discount,
        v_delivery_fee,
        'CONFIRMED',
        'PENDING',
        p_payment_method,
        p_shipping_address,
        case when p_promo_code is not null and trim(p_promo_code) != '' then upper(trim(p_promo_code)) else null end,
        p_notes,
        p_guest_access_token_hash,
        p_guest_access_expires_at,
        pg_catalog.now(),
        pg_catalog.now()
      )
      returning id, created_at into v_order_id, v_created_at;

      v_order_inserted := true;
      exit;
    exception when unique_violation then
      -- Inspect the exact failing constraint name to ensure ONLY order_number collisions trigger retry.
      -- Any other unique constraint violation (e.g. guest token hash or payment id) must NOT be suppressed.
      get stacked diagnostics v_constraint_name = constraint_name;
      if v_constraint_name = 'orders_order_number_key' or v_constraint_name like '%order_number%' then
        -- Generate a new cryptographic suffix using full unhyphenated pg_catalog.gen_random_uuid() and retry
        v_order_number := 'ORD-' ||
          pg_catalog.to_char(pg_catalog.now() at time zone 'Asia/Kolkata', 'YYMMDD') || '-' ||
          pg_catalog.upper(pg_catalog.replace(pg_catalog.gen_random_uuid()::text, '-', ''));
      else
        -- Re-throw any other unexpected unique constraint violation immediately, aborting transaction
        raise;
      end if;
    end;
  end loop;

  if not v_order_inserted then
    raise exception '%', jsonb_build_object(
      'code', 'CHECKOUT_TRANSACTION_FAILED',
      'message', 'Failed to allocate unique order number after retries.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 10. INSERT PUBLIC.ORDER_ITEMS RECORDS & ASSEMBLE ITEMS JSON
  -- ----------------------------------------------------------------------------
  for v_item in select * from jsonb_to_recordset(p_items) as x("serialNumber" text, "quantity" integer, "size" text) loop
    v_serial := upper(trim(v_item."serialNumber"));
    v_qty := v_item."quantity";

    select * into v_product
    from public.products
    where upper(trim(serial_number)) = v_serial;

    v_item_size := coalesce(
      nullif(trim(v_item."size"), ''),
      case when array_length(v_product.sizes, 1) > 0 then v_product.sizes[1] else 'One Size' end
    );

    insert into public.order_items (
      order_id,
      product_id,
      serial_number,
      name,
      price,
      quantity,
      size,
      color,
      image_url,
      slug
    )
    values (
      v_order_id,
      v_product.id,
      v_product.serial_number,
      v_product.name,
      v_product.price,
      v_qty,
      v_item_size,
      'Curated Wash',
      coalesce(case when array_length(v_product.images, 1) > 0 then v_product.images[1] else '' end),
      v_product.slug
    );

    v_items_json := v_items_json || jsonb_build_array(
      jsonb_build_object(
        'productId', v_product.id,
        'serialNumber', v_product.serial_number,
        'name', v_product.name,
        'price', v_product.price,
        'quantity', v_qty,
        'size', v_item_size,
        'image', coalesce(case when array_length(v_product.images, 1) > 0 then v_product.images[1] else '' end),
        'slug', v_product.slug
      )
    );
  end loop;

  -- ----------------------------------------------------------------------------
  -- 11. BUILD SANITIZED RESPONSE PAYLOAD & UPDATE IDEMPOTENCY RECORD
  -- Zero raw guest access token in response_payload or database
  -- ----------------------------------------------------------------------------
  v_response_payload := jsonb_build_object(
    'id', v_order_id,
    'orderNumber', v_order_number,
    'customerId', p_customer_id,
    'customer', jsonb_build_object(
      'name', trim(p_customer->>'name'),
      'email', lower(trim(p_customer->>'email')),
      'phone', trim(p_customer->>'phone')
    ),
    'shippingAddress', p_shipping_address,
    'items', v_items_json,
    'subtotal', v_subtotal,
    'discount', v_discount,
    'deliveryFee', v_delivery_fee,
    'totalAmount', v_total_amount,
    'status', 'CONFIRMED',
    'paymentMethod', p_payment_method,
    'paymentStatus', 'PENDING',
    'promoCode', case when p_promo_code is not null and trim(p_promo_code) != '' then upper(trim(p_promo_code)) else null end,
    'notes', p_notes,
    'guestAccessToken', null, -- Plaintext guest token is NEVER stored in database or cached payload
    'createdAt', v_created_at
  );

  update public.checkout_idempotency
  set response_payload = v_response_payload,
      order_number = v_order_number,
      expires_at = pg_catalog.now() + interval '24 hours'
  where key_hash = p_key_hash;

  return v_response_payload;
end;
$$;

--- 20261125_coupon_checkout_limits.sql ---
-- Enforces the existing per_customer_limit field in the atomic checkout transaction.
create or replace function public.process_storefront_checkout(
  p_key_hash text,
  p_fingerprint text,
  p_customer_id uuid,
  p_customer jsonb,
  p_shipping_address jsonb,
  p_items jsonb,
  p_payment_method text,
  p_promo_code text default null,
  p_notes text default null,
  p_guest_access_token_hash text default null,
  p_guest_access_expires_at timestamptz default null,
  p_order_number text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_existing_idem record;
  v_conflicting_idem record;
  v_claimed_idem record;
  v_order_number text;
  v_order_id uuid;
  v_order_inserted boolean := false;
  v_created_at timestamptz;
  v_whatsapp_number text;
  v_cleaned_whatsapp text;
  v_is_whatsapp_verified boolean;
  v_constraint_name text;
  v_attempt integer;
  v_item record;
  v_serial text;
  v_qty integer;
  v_item_size text;
  v_product record;
  v_missing_serial text;
  v_subtotal bigint := 0;
  v_discount bigint := 0;
  v_delivery_fee bigint := 0;
  v_total_amount bigint := 0;
  v_discounted_subtotal bigint := 0;
  v_coupon record;
  v_coupon_customer_usage bigint := 0;
  v_new_stock integer;
  v_new_status text;
  v_items_json jsonb := '[]'::jsonb;
  v_response_payload jsonb;
  v_loop_count integer := 0;
  v_retry_insert boolean := true;
  v_serials text[] := array[]::text[];
  v_quantities jsonb := '{}'::jsonb;
  v_variant_match boolean;
  v_var record;
begin
  -- ----------------------------------------------------------------------------
  -- 1. INPUT SANITIZATION & IDEMPOTENCY KEY CHECK
  -- ----------------------------------------------------------------------------
  if p_key_hash is null or length(p_key_hash) != 64 then
    raise exception '%', jsonb_build_object(
      'code', 'IDEMPOTENCY_KEY_REQUIRED',
      'message', 'A valid 64-character SHA-256 idempotency key hash is required.'
    )::text;
  end if;

  if p_fingerprint is null or length(p_fingerprint) != 64 then
    raise exception '%', jsonb_build_object(
      'code', 'VALIDATION_ERROR',
      'message', 'A valid 64-character SHA-256 request fingerprint is required.'
    )::text;
  end if;

  -- Bounded retry loop for idempotency claim (handles races with background cleanup)
  while v_retry_insert and v_loop_count < 3 loop
    v_loop_count := v_loop_count + 1;
    v_retry_insert := false;

    select * into v_existing_idem
    from public.checkout_idempotency
    where key_hash = p_key_hash;

    if found then
      if v_existing_idem.expires_at > pg_catalog.now() then
        if v_existing_idem.fingerprint != p_fingerprint then
          raise exception '%', jsonb_build_object(
            'code', 'IDEMPOTENCY_KEY_REUSED',
            'message', 'This idempotency key was already used for a different checkout.'
          )::text;
        end if;

        if v_existing_idem.response_payload is not null then
          return v_existing_idem.response_payload;
        else
          raise exception '%', jsonb_build_object(
            'code', 'CHECKOUT_IN_PROGRESS',
            'message', 'Checkout with this key is already being processed. Retry shortly.'
          )::text;
        end if;
      else
        delete from public.checkout_idempotency
        where key_hash = p_key_hash
          and expires_at <= pg_catalog.now();
      end if;
    end if;

    -- Collision-resistant order number: ORD-YYMMDD-XXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXX (122 bits of CSPRNG entropy)
    -- Uses full unhyphenated pg_catalog.gen_random_uuid() to guarantee 100% resolution under empty search_path
    -- without external extension dependencies (pgcrypto or uuid-ossp).
    v_order_number := coalesce(
      nullif(trim(p_order_number), ''),
      'ORD-' ||
        pg_catalog.to_char(pg_catalog.now() at time zone 'Asia/Kolkata', 'YYMMDD') || '-' ||
        pg_catalog.upper(pg_catalog.replace(pg_catalog.gen_random_uuid()::text, '-', ''))
    );

    insert into public.checkout_idempotency (
      key_hash,
      fingerprint,
      order_number,
      response_payload,
      expires_at
    )
    values (
      p_key_hash,
      p_fingerprint,
      v_order_number,
      null,
      pg_catalog.now() + interval '5 minutes'
    )
    on conflict (key_hash) do nothing
    returning * into v_claimed_idem;

    if v_claimed_idem.id is null then
      select * into v_conflicting_idem
      from public.checkout_idempotency
      where key_hash = p_key_hash
      for update;

      if not found then
        v_retry_insert := true;
      else
        if v_conflicting_idem.expires_at > pg_catalog.now() then
          if v_conflicting_idem.fingerprint != p_fingerprint then
            raise exception '%', jsonb_build_object(
              'code', 'IDEMPOTENCY_KEY_REUSED',
              'message', 'This idempotency key was already used for a different checkout.'
            )::text;
          elsif v_conflicting_idem.response_payload is not null then
            return v_conflicting_idem.response_payload;
          else
            raise exception '%', jsonb_build_object(
              'code', 'CHECKOUT_IN_PROGRESS',
              'message', 'Checkout with this key is already being processed. Retry shortly.'
            )::text;
          end if;
        else
          delete from public.checkout_idempotency
          where key_hash = p_key_hash;
          v_retry_insert := true;
        end if;
      end if;
    end if;
  end loop;

  if v_claimed_idem.id is null and v_response_payload is null then
    raise exception '%', jsonb_build_object(
      'code', 'CHECKOUT_IN_PROGRESS',
      'message', 'Checkout reservation could not be acquired. Please retry.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 2. VALIDATE PAYMENT METHOD & WHATSAPP CONFIGURATION
  -- ----------------------------------------------------------------------------
  if p_payment_method = 'ONLINE_UPI' then
    raise exception '%', jsonb_build_object(
      'code', 'PAYMENT_PROVIDER_UNAVAILABLE',
      'message', 'Online payment is temporarily unavailable. Please try again later.'
    )::text;
  elsif p_payment_method not in ('CASH_ON_DELIVERY', 'WHATSAPP_UPI') then
    raise exception '%', jsonb_build_object(
      'code', 'VALIDATION_ERROR',
      'message', 'Invalid payment method: must be CASH_ON_DELIVERY or WHATSAPP_UPI.'
    )::text;
  end if;

  if p_payment_method = 'WHATSAPP_UPI' then
    select whatsapp_number, is_whatsapp_verified
    into v_whatsapp_number, v_is_whatsapp_verified
    from public.site_settings
    limit 1;

    v_cleaned_whatsapp := pg_catalog.regexp_replace(coalesce(v_whatsapp_number, ''), '\D', '', 'g');

    -- WhatsApp checkout requires an explicit store-verified configuration.
    -- Syntactic validity or absence of blacklisted numbers is NOT sufficient proof of store ownership.
    if coalesce(v_is_whatsapp_verified, false) is not true or
       v_cleaned_whatsapp is null or
       not (v_cleaned_whatsapp ~ '^[1-9][0-9]{9,14}$') or
       v_cleaned_whatsapp in ('919876543210', '910000000000', '1234567890', '9876543210') or
       v_cleaned_whatsapp like '%000000%' then
      raise exception '%', jsonb_build_object(
        'code', 'PAYMENT_METHOD_UNAVAILABLE',
        'message', 'WhatsApp / UPI payment is temporarily unavailable because store contact details have not been verified.'
      )::text;
    end if;
  end if;

  -- ----------------------------------------------------------------------------
  -- 3. VALIDATE CUSTOMER & SHIPPING ADDRESS
  -- ----------------------------------------------------------------------------
  if p_customer is null or
     trim(coalesce(p_customer->>'name', '')) = '' or
     trim(coalesce(p_customer->>'email', '')) = '' or
     trim(coalesce(p_customer->>'phone', '')) = '' then
    raise exception '%', jsonb_build_object(
      'code', 'MISSING_CUSTOMER_INFO',
      'message', 'Customer name, email, and phone are required.'
    )::text;
  end if;

  if p_shipping_address is null or
     trim(coalesce(p_shipping_address->>'addressLine', '')) = '' or
     trim(coalesce(p_shipping_address->>'city', '')) = '' or
     trim(coalesce(p_shipping_address->>'state', '')) = '' or
     trim(coalesce(p_shipping_address->>'postalCode', '')) = '' then
    raise exception '%', jsonb_build_object(
      'code', 'INCOMPLETE_ADDRESS',
      'message', 'Complete delivery address (line, city, state, postal code) is required.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 4. PARSE & AGGREGATE ITEMS
  -- ----------------------------------------------------------------------------
  if p_items is null or jsonb_array_length(p_items) = 0 then
    raise exception '%', jsonb_build_object(
      'code', 'EMPTY_ORDER_ITEMS',
      'message', 'Order must contain at least one item.'
    )::text;
  end if;

  for v_item in select * from jsonb_to_recordset(p_items) as x("serialNumber" text, "quantity" integer, "size" text) loop
    v_serial := upper(trim(coalesce(v_item."serialNumber", '')));
    v_qty := coalesce(v_item."quantity", 0);

    if v_serial = '' then
      raise exception '%', jsonb_build_object(
        'code', 'VALIDATION_ERROR',
        'message', 'Item serial number is required.'
      )::text;
    end if;

    if v_qty <= 0 or v_qty > 10 then
      raise exception '%', jsonb_build_object(
        'code', 'INVALID_QUANTITY',
        'message', 'Quantity must be between 1 and 10 per item.'
      )::text;
    end if;

    if not (v_serials @> array[v_serial]) then
      v_serials := array_append(v_serials, v_serial);
      v_quantities := v_quantities || jsonb_build_object(v_serial, v_qty);
    else
      v_qty := (v_quantities->>v_serial)::integer + v_qty;
      if v_qty > 10 then
        raise exception '%', jsonb_build_object(
          'code', 'INVALID_QUANTITY',
          'message', 'Maximum quantity per product is 10.'
        )::text;
      end if;
      v_quantities := v_quantities || jsonb_build_object(v_serial, v_qty);
    end if;
  end loop;

  -- ----------------------------------------------------------------------------
  -- 5. LOCK PRODUCT ROWS IN DETERMINISTIC SORT ORDER & VALIDATE VARIANTS
  -- ----------------------------------------------------------------------------
  for v_product in
    select p.*
    from public.products p
    where upper(trim(p.serial_number)) = any(v_serials)
    order by p.serial_number asc
    for update
  loop
    v_serial := upper(trim(v_product.serial_number));
    v_qty := (v_quantities->>v_serial)::integer;

    -- Stock exhaustion check takes priority so items decremented to 0 report OUT_OF_STOCK
    if v_product.stock is null or v_product.stock < v_qty then
      raise exception '%', jsonb_build_object(
        'code', 'OUT_OF_STOCK',
        'message', 'Insufficient stock for "' || v_product.serial_number || '".'
      )::text;
    end if;

    if v_product.status not in ('AVAILABLE', 'ACTIVE', 'PUBLISHED') then
      raise exception '%', jsonb_build_object(
        'code', 'PRODUCT_NOT_ORDERABLE',
        'message', 'Product "' || v_product.serial_number || '" is not available to order.'
      )::text;
    end if;

    if v_product.price is null or v_product.price < 0 or v_product.price > 1000000000 then
      raise exception '%', jsonb_build_object(
        'code', 'INVALID_PRODUCT_PRICE',
        'message', 'Product "' || v_product.serial_number || '" has an invalid stored price.'
      )::text;
    end if;

    v_subtotal := v_subtotal + (v_product.price * v_qty);
  end loop;

  if (
    select count(*)
    from public.products p
    where upper(trim(p.serial_number)) = any(v_serials)
  ) < array_length(v_serials, 1) then
    select s into v_missing_serial
    from unnest(v_serials) s
    where not exists (
      select 1 from public.products p where upper(trim(p.serial_number)) = s
    )
    limit 1;

    raise exception '%', jsonb_build_object(
      'code', 'PRODUCT_NOT_ORDERABLE',
      'message', 'Product "' || coalesce(v_missing_serial, 'UNKNOWN') || '" is not available to order.'
    )::text;
  end if;

  -- Validate sizes/variants for each requested item
  for v_item in select * from jsonb_to_recordset(p_items) as x("serialNumber" text, "quantity" integer, "size" text) loop
    v_serial := upper(trim(v_item."serialNumber"));
    v_item_size := trim(coalesce(v_item."size", ''));

    select * into v_product
    from public.products
    where upper(trim(serial_number)) = v_serial;

    -- If product defines available sizes:
    if array_length(v_product.sizes, 1) > 0 then
      if v_item_size = '' then
        -- If single size option exists, default to it
        if array_length(v_product.sizes, 1) = 1 then
          v_item_size := v_product.sizes[1];
        else
          raise exception '%', jsonb_build_object(
            'code', 'INVALID_VARIANT',
            'message', 'Size is required for product "' || v_product.serial_number || '". Available: ' || array_to_string(v_product.sizes, ', ') || '.'
          )::text;
        end if;
      elsif not (v_product.sizes @> array[v_item_size]) then
        -- Check if size exists in variants jsonb
        v_variant_match := false;
        if jsonb_array_length(v_product.variants) > 0 then
          for v_var in select * from jsonb_to_recordset(v_product.variants) as z("size" text, "label" text) loop
            if upper(trim(coalesce(v_var."size", ''))) = upper(v_item_size) or upper(trim(coalesce(v_var."label", ''))) = upper(v_item_size) then
              v_variant_match := true;
              exit;
            end if;
          end loop;
        end if;

        if not v_variant_match then
          raise exception '%', jsonb_build_object(
            'code', 'INVALID_VARIANT',
            'message', 'Size "' || v_item_size || '" is not available for product "' || v_product.serial_number || '". Available sizes: ' || array_to_string(v_product.sizes, ', ') || '.'
          )::text;
        end if;
      end if;
    elsif jsonb_array_length(v_product.variants) > 0 then
      -- Product uses variants jsonb exclusively
      v_variant_match := false;
      for v_var in select * from jsonb_to_recordset(v_product.variants) as z("size" text, "label" text) loop
        if v_item_size != '' and (upper(trim(coalesce(v_var."size", ''))) = upper(v_item_size) or upper(trim(coalesce(v_var."label", ''))) = upper(v_item_size)) then
          v_variant_match := true;
          exit;
        end if;
      end loop;

      if v_item_size != '' and not v_variant_match then
        raise exception '%', jsonb_build_object(
          'code', 'INVALID_VARIANT',
          'message', 'Variant "' || v_item_size || '" is not available for product "' || v_product.serial_number || '".'
        )::text;
      end if;
    end if;
  end loop;

  if v_subtotal < 0 or v_subtotal > 10000000000 then
    raise exception '%', jsonb_build_object(
      'code', 'INVALID_ORDER_TOTAL',
      'message', 'Order subtotal exceeds supported currency range.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 6. VALIDATE & LOCK COUPON (REJECT UNSUPPORTED TYPES EXPLICITLY)
  -- ----------------------------------------------------------------------------
  if p_promo_code is not null and trim(p_promo_code) != '' then
    select * into v_coupon
    from public.coupons
    where upper(trim(code)) = upper(trim(p_promo_code))
    for update;

    if not found or not v_coupon.is_active then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'Invalid or expired promotional code.'
      )::text;
    end if;

    -- Reject unsupported coupon types BEFORE updating usage count
    if v_coupon.discount_type not in ('PERCENTAGE', 'FLAT') then
      raise exception '%', jsonb_build_object(
        'code', 'UNSUPPORTED_COUPON_TYPE',
        'message', 'Promo code discount type "' || v_coupon.discount_type || '" is not supported at checkout.'
      )::text;
    end if;

    if v_coupon.starts_at is not null and v_coupon.starts_at > pg_catalog.now() then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'This promo code is not yet active.'
      )::text;
    end if;

    if v_coupon.ends_at is not null and v_coupon.ends_at < pg_catalog.now() then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'This promo code has expired.'
      )::text;
    end if;

    if v_coupon.usage_limit is not null and v_coupon.usage_count >= v_coupon.usage_limit then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'This promo code has reached its maximum usage limit.'
      )::text;
    end if;

    if v_coupon.per_customer_limit is not null then
      select count(*) into v_coupon_customer_usage
      from public.orders as prior_order
      where upper(trim(prior_order.promo_code)) = upper(trim(v_coupon.code))
        and (
          (p_customer_id is not null and prior_order.customer_id = p_customer_id)
          or lower(prior_order.customer_email) = lower(trim(p_customer->>'email'))
        );
      if v_coupon_customer_usage >= v_coupon.per_customer_limit then
        raise exception '%', jsonb_build_object(
          'code', 'COUPON_REQUIRED',
          'message', 'This promo code has reached its per-customer usage limit.'
        )::text;
      end if;
    end if;

    if v_subtotal < v_coupon.min_order_paise then
      raise exception '%', jsonb_build_object(
        'code', 'COUPON_REQUIRED',
        'message', 'Code ' || v_coupon.code || ' requires a minimum bag value of ₹' ||
          (round(v_coupon.min_order_paise / 100))::text || '.'
      )::text;
    end if;

    -- Accurate numeric percentage calculation avoiding integer division truncation
    if v_coupon.discount_type = 'PERCENTAGE' then
      v_discount := round((v_subtotal::numeric * v_coupon.discount_value::numeric) / 100.0)::bigint;
      if v_coupon.max_discount_paise is not null and v_discount > v_coupon.max_discount_paise then
        v_discount := v_coupon.max_discount_paise;
      end if;
      v_discount := least(v_discount, v_subtotal);
    elsif v_coupon.discount_type = 'FLAT' then
      v_discount := least(v_coupon.discount_value, v_subtotal);
    else
      v_discount := 0;
    end if;

    -- Increment coupon usage atomically
    update public.coupons
    set usage_count = usage_count + 1,
        updated_at = pg_catalog.now()
    where id = v_coupon.id;
  end if;

  -- ----------------------------------------------------------------------------
  -- 7. DELIVERY FEE & TOTAL CALCULATION (CANONICAL PAISE: 1 INR = 100 PAISE)
  -- ----------------------------------------------------------------------------
  v_discounted_subtotal := v_subtotal - v_discount;
  if v_discounted_subtotal >= 150000 then
    v_delivery_fee := 0; -- Free shipping threshold on bag value >= ₹1,500
  else
    v_delivery_fee := 9000; -- Standard ₹90 delivery fee
  end if;

  v_total_amount := v_discounted_subtotal + v_delivery_fee;

  if v_total_amount < 0 or v_total_amount > 10000000000 then
    raise exception '%', jsonb_build_object(
      'code', 'INVALID_ORDER_TOTAL',
      'message', 'Order total exceeds supported currency range.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 8. DECREMENT INVENTORY EXACTLY ONCE
  -- ----------------------------------------------------------------------------
  for v_product in
    select p.*
    from public.products p
    where upper(trim(p.serial_number)) = any(v_serials)
    order by p.serial_number asc
  loop
    v_serial := upper(trim(v_product.serial_number));
    v_qty := (v_quantities->>v_serial)::integer;
    v_new_stock := v_product.stock - v_qty;
    v_new_status := case when v_new_stock = 0 then 'SOLD_OUT' else v_product.status end;

    update public.products
    set stock = v_new_stock,
        status = v_new_status,
        updated_at = pg_catalog.now()
    where id = v_product.id;
  end loop;

  -- ----------------------------------------------------------------------------
  -- 9. INSERT PUBLIC.ORDERS RECORD (WITH RETRY ON UNIQUE ORDER NUMBER COLLISION)
  -- ----------------------------------------------------------------------------
  for v_attempt in 1..5 loop
    begin
      insert into public.orders (
        order_number,
        customer_id,
        customer_name,
        customer_email,
        customer_phone,
        total_amount,
        subtotal,
        discount_amount,
        shipping_amount,
        status,
        payment_status,
        payment_method,
        shipping_address,
        promo_code,
        notes,
        guest_access_token_hash,
        guest_access_expires_at,
        created_at,
        updated_at
      )
      values (
        v_order_number,
        p_customer_id,
        trim(p_customer->>'name'),
        lower(trim(p_customer->>'email')),
        trim(p_customer->>'phone'),
        v_total_amount,
        v_subtotal,
        v_discount,
        v_delivery_fee,
        'CONFIRMED',
        'PENDING',
        p_payment_method,
        p_shipping_address,
        case when p_promo_code is not null and trim(p_promo_code) != '' then upper(trim(p_promo_code)) else null end,
        p_notes,
        p_guest_access_token_hash,
        p_guest_access_expires_at,
        pg_catalog.now(),
        pg_catalog.now()
      )
      returning id, created_at into v_order_id, v_created_at;

      v_order_inserted := true;
      exit;
    exception when unique_violation then
      -- Inspect the exact failing constraint name to ensure ONLY order_number collisions trigger retry.
      -- Any other unique constraint violation (e.g. guest token hash or payment id) must NOT be suppressed.
      get stacked diagnostics v_constraint_name = constraint_name;
      if v_constraint_name = 'orders_order_number_key' or v_constraint_name like '%order_number%' then
        -- Generate a new cryptographic suffix using full unhyphenated pg_catalog.gen_random_uuid() and retry
        v_order_number := 'ORD-' ||
          pg_catalog.to_char(pg_catalog.now() at time zone 'Asia/Kolkata', 'YYMMDD') || '-' ||
          pg_catalog.upper(pg_catalog.replace(pg_catalog.gen_random_uuid()::text, '-', ''));
      else
        -- Re-throw any other unexpected unique constraint violation immediately, aborting transaction
        raise;
      end if;
    end;
  end loop;

  if not v_order_inserted then
    raise exception '%', jsonb_build_object(
      'code', 'CHECKOUT_TRANSACTION_FAILED',
      'message', 'Failed to allocate unique order number after retries.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 10. INSERT PUBLIC.ORDER_ITEMS RECORDS & ASSEMBLE ITEMS JSON
  -- ----------------------------------------------------------------------------
  for v_item in select * from jsonb_to_recordset(p_items) as x("serialNumber" text, "quantity" integer, "size" text) loop
    v_serial := upper(trim(v_item."serialNumber"));
    v_qty := v_item."quantity";

    select * into v_product
    from public.products
    where upper(trim(serial_number)) = v_serial;

    v_item_size := coalesce(
      nullif(trim(v_item."size"), ''),
      case when array_length(v_product.sizes, 1) > 0 then v_product.sizes[1] else 'One Size' end
    );

    insert into public.order_items (
      order_id,
      product_id,
      serial_number,
      name,
      price,
      quantity,
      size,
      color,
      image_url,
      slug
    )
    values (
      v_order_id,
      v_product.id,
      v_product.serial_number,
      v_product.name,
      v_product.price,
      v_qty,
      v_item_size,
      'Curated Wash',
      coalesce(case when array_length(v_product.images, 1) > 0 then v_product.images[1] else '' end),
      v_product.slug
    );

    v_items_json := v_items_json || jsonb_build_array(
      jsonb_build_object(
        'productId', v_product.id,
        'serialNumber', v_product.serial_number,
        'name', v_product.name,
        'price', v_product.price,
        'quantity', v_qty,
        'size', v_item_size,
        'image', coalesce(case when array_length(v_product.images, 1) > 0 then v_product.images[1] else '' end),
        'slug', v_product.slug
      )
    );
  end loop;

  -- ----------------------------------------------------------------------------
  -- 11. BUILD SANITIZED RESPONSE PAYLOAD & UPDATE IDEMPOTENCY RECORD
  -- Zero raw guest access token in response_payload or database
  -- ----------------------------------------------------------------------------
  v_response_payload := jsonb_build_object(
    'id', v_order_id,
    'orderNumber', v_order_number,
    'customerId', p_customer_id,
    'customer', jsonb_build_object(
      'name', trim(p_customer->>'name'),
      'email', lower(trim(p_customer->>'email')),
      'phone', trim(p_customer->>'phone')
    ),
    'shippingAddress', p_shipping_address,
    'items', v_items_json,
    'subtotal', v_subtotal,
    'discount', v_discount,
    'deliveryFee', v_delivery_fee,
    'totalAmount', v_total_amount,
    'status', 'CONFIRMED',
    'paymentMethod', p_payment_method,
    'paymentStatus', 'PENDING',
    'promoCode', case when p_promo_code is not null and trim(p_promo_code) != '' then upper(trim(p_promo_code)) else null end,
    'notes', p_notes,
    'guestAccessToken', null, -- Plaintext guest token is NEVER stored in database or cached payload
    'createdAt', v_created_at
  );

  update public.checkout_idempotency
  set response_payload = v_response_payload,
      order_number = v_order_number,
      expires_at = pg_catalog.now() + interval '24 hours'
  where key_hash = p_key_hash;

  return v_response_payload;
end;
$$;

-- Existing row triggers snapshot whole records; strip credentials and customer
-- identifiers recursively before those snapshots are stored in the audit trail.
create or replace function public.fn_redact_audit_json(value jsonb)
returns jsonb
language plpgsql
immutable
as $$
declare
  result jsonb;
begin
  if value is null then
    return null;
  elsif jsonb_typeof(value) = 'object' then
    select coalesce(jsonb_object_agg(entry.key, public.fn_redact_audit_json(entry.value)), '{}'::jsonb)
      into result
      from jsonb_each(value) as entry(key, value)
      where not (
        regexp_replace(lower(entry.key), '[^a-z0-9]', '', 'g') like '%password%'
        or regexp_replace(lower(entry.key), '[^a-z0-9]', '', 'g') like '%token%'
        or regexp_replace(lower(entry.key), '[^a-z0-9]', '', 'g') like '%secret%'
        or regexp_replace(lower(entry.key), '[^a-z0-9]', '', 'g') like '%authorization%'
        or regexp_replace(lower(entry.key), '[^a-z0-9]', '', 'g') like '%cookie%'
        or regexp_replace(lower(entry.key), '[^a-z0-9]', '', 'g') like '%servicerole%'
        or regexp_replace(lower(entry.key), '[^a-z0-9]', '', 'g') like '%apikey%'
        or regexp_replace(lower(entry.key), '[^a-z0-9]', '', 'g') like 'customer%'
        or regexp_replace(lower(entry.key), '[^a-z0-9]', '', 'g') in ('shippingaddress', 'billingaddress')
      );
    return result;
  elsif jsonb_typeof(value) = 'array' then
    select coalesce(jsonb_agg(public.fn_redact_audit_json(entry.value) order by entry.ordinality), '[]'::jsonb)
      into result
      from jsonb_array_elements(value) with ordinality as entry(value, ordinality);
    return result;
  end if;

  return value;
end;
$$;

create or replace function public.fn_activity_audit_log()
returns trigger
language plpgsql
security definer
as $$
declare
  rec_id text;
  old_data jsonb;
  new_data jsonb;
  act_type text;
begin
  if tg_op = 'DELETE' then
    rec_id := coalesce(old.id::text, '');
    old_data := public.fn_redact_audit_json(to_jsonb(old));
    new_data := null;
    act_type := 'DELETE';
  elsif tg_op = 'INSERT' then
    rec_id := coalesce(new.id::text, '');
    old_data := null;
    new_data := public.fn_redact_audit_json(to_jsonb(new));
    act_type := 'INSERT';
  else
    rec_id := coalesce(new.id::text, '');
    old_data := public.fn_redact_audit_json(to_jsonb(old));
    new_data := public.fn_redact_audit_json(to_jsonb(new));
    if old_data->>'status' is distinct from new_data->>'status' then
      act_type := 'STATUS_CHANGE';
    else
      act_type := 'UPDATE';
    end if;
  end if;

  insert into public.activity_audit_logs (
    actor_id, actor_role, actor_email, table_name, record_id, action,
    before_state, after_state
  ) values (
    coalesce(auth.uid()::text, 'service_role'),
    coalesce(auth.jwt() -> 'app_metadata' ->> 'role', 'admin'),
    auth.jwt() ->> 'email', tg_table_name, rec_id, act_type, old_data, new_data
  );

  return coalesce(new, old);
end;
$$;

-- Atomic admin order status, cancellation/inventory, and payment operations.
-- Express authenticates and authorizes admins. These SECURITY DEFINER RPCs are
-- executable only by service_role; actor values are supplied by that trusted API.

create or replace function public.admin_update_order_status(
  p_order_ref text,
  p_new_status text,
  p_actor_id text,
  p_actor_email text,
  p_actor_role text,
  p_ip_address text default null,
  p_user_agent text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order public.orders%rowtype;
  v_old_status text;
  v_item record;
  v_product_id uuid;
  v_items jsonb;
begin
  if coalesce(auth.role(), '') <> 'service_role' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Admin order operation is not authorized.')::text using errcode = '42501';
  end if;
  if p_order_ref is null or length(trim(p_order_ref)) not between 8 and 100
     or not (trim(p_order_ref) ~* '^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$'
             or upper(trim(p_order_ref)) ~ '^ORD-[A-Z0-9-]{8,80}$')
     or p_new_status is null or p_new_status not in ('PENDING', 'CONFIRMED', 'SHIPPED', 'DELIVERED', 'CANCELLED')
     or p_actor_id is null or length(trim(p_actor_id)) = 0
     or p_actor_role is null or p_actor_role not in ('owner', 'staff') then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Admin order status request is invalid.')::text using errcode = '22023';
  end if;

  select o.* into v_order
  from public.orders as o
  where o.order_number = upper(trim(p_order_ref)) or o.id::text = trim(p_order_ref)
  for update;
  if not found then
    raise exception '%', jsonb_build_object('code', 'ORDER_NOT_FOUND', 'message', 'Order was not found.')::text using errcode = 'P0002';
  end if;

  v_old_status := v_order.status;
  if not (
    (v_old_status = 'PENDING' and p_new_status in ('CONFIRMED', 'CANCELLED'))
    or (v_old_status = 'CONFIRMED' and p_new_status in ('SHIPPED', 'CANCELLED'))
    or (v_old_status = 'SHIPPED' and p_new_status in ('DELIVERED', 'CANCELLED'))
  ) then
    raise exception '%', jsonb_build_object('code', 'INVALID_STATUS_TRANSITION', 'message', 'This order status transition is not allowed.')::text using errcode = 'P0001';
  end if;

  if p_new_status = 'CANCELLED' then
    -- Aggregate duplicate order lines, then lock products in stable serial order.
    for v_item in
      select oi.serial_number, sum(oi.quantity)::integer as quantity
      from public.order_items as oi
      where oi.order_id = v_order.id
      group by oi.serial_number
      order by oi.serial_number
    loop
      if v_item.serial_number is null or v_item.quantity <= 0 then
        raise exception '%', jsonb_build_object('code', 'ORDER_INVENTORY_UNAVAILABLE', 'message', 'Order inventory could not be restored.')::text using errcode = 'P0001';
      end if;
      select p.id into v_product_id
      from public.products as p
      where p.serial_number = upper(v_item.serial_number)
      for update;
      if not found then
        raise exception '%', jsonb_build_object('code', 'ORDER_INVENTORY_UNAVAILABLE', 'message', 'A product required for inventory restoration no longer exists.')::text using errcode = 'P0001';
      end if;
      update public.products as p
      set stock = p.stock + v_item.quantity,
          status = 'ACTIVE',
          updated_at = timezone('utc'::text, now())
      where p.id = v_product_id;
    end loop;
  end if;

  update public.orders as o
  set status = p_new_status,
      updated_at = timezone('utc'::text, now())
  where o.id = v_order.id
  returning o.* into v_order;

  -- Preserve the existing explicit ORDER_STATUS_UPDATE audit event. The table
  -- trigger also records its generic status snapshot as it does for other updates.
  insert into public.activity_audit_logs (
    actor_id, actor_email, actor_role, table_name, record_id, action,
    after_state, ip_address, user_agent
  ) values (
    p_actor_id, p_actor_email, p_actor_role, 'Order', v_order.order_number,
    'ORDER_STATUS_UPDATE',
    jsonb_build_object('previousStatus', v_old_status, 'newStatus', p_new_status),
    coalesce(p_ip_address, '127.0.0.1'), left(coalesce(p_user_agent, 'CircleUp Admin'), 1024)
  );

  select coalesce(jsonb_agg(to_jsonb(oi) order by oi.id), '[]'::jsonb)
  into v_items from public.order_items as oi where oi.order_id = v_order.id;
  return jsonb_build_object('order', to_jsonb(v_order), 'order_items', v_items);
end;
$$;

create or replace function public.admin_update_user(

language plpgsql
security invoker
set search_path = public, pg_temp
as $$

begin
  if p_patch is null or jsonb_typeof(p_patch) <> 'object' or
     (p_patch - array['email', 'name', 'role', 'is_active', 'password_hash']) <> '{}'::jsonb then
    raise exception using errcode = '22023', message = 'INVALID_USER_PATCH';
  end if;
  if p_patch ? 'email' and (length(trim(p_patch->>'email')) > 254 or (p_patch->>'email') !~* '^[^[:space:]@]+@[^[:space:]@]+\.[^[:space:]@]+$') then
    raise exception using errcode = '22023', message = 'INVALID_EMAIL';
  end if;

    raise exception using errcode = '22023', message = 'INVALID_NAME';
  end if;

    raise exception using errcode = '22023', message = 'INVALID_ROLE';
  end if;
  if p_patch ? 'is_active' and jsonb_typeof(p_patch->'is_active') <> 'boolean' then
    raise exception using errcode = '22023', message = 'INVALID_ACTIVE_STATE';
  end if;

    raise exception using errcode = '22023', message = 'INVALID_PASSWORD_HASH';
  end if;

  end if;

  if not found then raise exception using errcode = 'P0002', message = 'USER_NOT_FOUND'; end if;
  if p_actor_id = p_target_id and ((p_patch ? 'role' and p_patch->>'role' <> target.role) or (p_patch->>'is_active' = 'false')) then
    raise exception using errcode = '42501', message = 'SELF_ADMIN_CHANGE_FORBIDDEN';
  end if;
  if target.role = 'owner' and target.is_active and
    ((p_patch->>'role' = 'staff') or (p_patch->>'is_active' = 'false')) then
    perform id from public.admin_users where role = 'owner' and is_active = true for update;
    select count(*) into owner_count from public.admin_users where role = 'owner' and is_active = true;
    if owner_count <= 1 then raise exception using errcode = '23514', message = 'LAST_OWNER_REQUIRED'; end if;
  end if;
  update public.admin_users set

    full_name = case when p_patch ? 'name' then p_patch->>'name' else full_name end,
    role = case when p_patch ? 'role' then p_patch->>'role' else role end,
    is_active = case when p_patch ? 'is_active' then (p_patch->>'is_active')::boolean else is_active end,
    password_hash = case when p_patch ? 'password_hash' then p_patch->>'password_hash' else password_hash end,
    updated_at = now()
    where id = p_target_id returning * into target;
  if p_patch->>'is_active' = 'false' or p_patch ? 'role' or p_patch ? 'password_hash' then
    update public.admin_auth_sessions set revoked_at = now() where user_id = p_target_id and revoked_at is null;
  end if;
  return jsonb_build_object(
    'id', target.id,
    'email', target.email,
    'full_name', target.full_name,
    'role', target.role,
    'is_active', target.is_active,
    'last_login_at', target.last_login_at,
    'created_at', target.created_at,
    'updated_at', target.updated_at
  );
end;
$$;

     or p_new_status is null or p_new_status not in ('PENDING', 'CONFIRMED', 'SHIPPED', 'DELIVERED', 'CANCELLED')
     or p_actor_id is null or length(trim(p_actor_id)) = 0
     or p_actor_role is null or p_actor_role not in ('owner', 'staff') then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Admin order status request is invalid.')::text using errcode = '22023';
  end if;

  select o.* into v_order
  from public.orders as o
  where o.order_number = upper(trim(p_order_ref)) or o.id::text = trim(p_order_ref)
  for update;
  if not found then
    raise exception '%', jsonb_build_object('code', 'ORDER_NOT_FOUND', 'message', 'Order was not found.')::text using errcode = 'P0002';
  end if;

  v_old_status := v_order.status;
  if not (
    (v_old_status = 'PENDING' and p_new_status in ('CONFIRMED', 'CANCELLED'))
    or (v_old_status = 'CONFIRMED' and p_new_status in ('SHIPPED', 'CANCELLED'))
    or (v_old_status = 'SHIPPED' and p_new_status in ('DELIVERED', 'CANCELLED'))
  ) then
    raise exception '%', jsonb_build_object('code', 'INVALID_STATUS_TRANSITION', 'message', 'This order status transition is not allowed.')::text using errcode = 'P0001';
  end if;

  if p_new_status = 'CANCELLED' then
    -- Aggregate duplicate order lines, then lock products in stable serial order.
    for v_item in
      select oi.serial_number, sum(oi.quantity)::integer as quantity
      from public.order_items as oi
      where oi.order_id = v_order.id
      group by oi.serial_number
      order by oi.serial_number
    loop
      if v_item.serial_number is null or v_item.quantity <= 0 then
        raise exception '%', jsonb_build_object('code', 'ORDER_INVENTORY_UNAVAILABLE', 'message', 'Order inventory could not be restored.')::text using errcode = 'P0001';
      end if;
      select p.id into v_product_id
      from public.products as p
      where p.serial_number = upper(v_item.serial_number)
      for update;
      if not found then
        raise exception '%', jsonb_build_object('code', 'ORDER_INVENTORY_UNAVAILABLE', 'message', 'A product required for inventory restoration no longer exists.')::text using errcode = 'P0001';
      end if;
      update public.products as p
      set stock = p.stock + v_item.quantity,
          status = case when p.status = 'SOLD_OUT' then 'ACTIVE' else p.status end,
          updated_at = timezone('utc'::text, now())
      where p.id = v_product_id;
    end loop;
  end if;

  update public.orders as o
  set status = p_new_status,
      updated_at = timezone('utc'::text, now())
  where o.id = v_order.id
  returning o.* into v_order;

  -- Preserve the existing explicit ORDER_STATUS_UPDATE audit event. The table
  -- trigger also records its generic status snapshot as it does for other updates.
  insert into public.activity_audit_logs (
    actor_id, actor_email, actor_role, table_name, record_id, action,
    after_state, ip_address, user_agent
  ) values (
    p_actor_id, p_actor_email, p_actor_role, 'Order', v_order.order_number,
    'ORDER_STATUS_UPDATE',
    jsonb_build_object('previousStatus', v_old_status, 'newStatus', p_new_status),
    coalesce(p_ip_address, '127.0.0.1'), left(coalesce(p_user_agent, 'CircleUp Admin'), 1024)
  );

  select coalesce(jsonb_agg(to_jsonb(oi) order by oi.id), '[]'::jsonb)
  into v_items from public.order_items as oi where oi.order_id = v_order.id;
  return jsonb_build_object('order', to_jsonb(v_order), 'order_items', v_items);
end;

-- ------------------------------------------------------------------------------
-- 3. Canonical admin_update_order_status RPC (Resolves P0-3)
-- Enforces canonical transition matrix across all 10 states and handles inventory
-- ------------------------------------------------------------------------------
create or replace function public.admin_update_order_status(
  p_order_ref text,
  p_new_status text,
  p_actor_id text,
  p_actor_email text,
  p_actor_role text,
  p_ip_address text default null,
  p_user_agent text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order public.orders%rowtype;
  v_old_status text;
  v_item record;
  v_product_id uuid;
  v_items jsonb;
  v_allowed boolean := false;
begin
  if coalesce(auth.role(), '') <> 'service_role' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Admin order operation is not authorized.')::text using errcode = '42501';
  end if;

  if p_order_ref is null or length(trim(p_order_ref)) not between 8 and 100
     or not (trim(p_order_ref) ~* '^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$'
             or upper(trim(p_order_ref)) ~ '^ORD-[A-Z0-9-]{8,80}$')
     or p_new_status is null or p_new_status not in (
       'PENDING', 'CONFIRMED', 'PACKED', 'SHIPPED', 'OUT_FOR_DELIVERY',
       'DELIVERED', 'CANCELLED', 'RETURN_REQUESTED', 'RETURNED', 'REFUNDED'
     )
     or p_actor_id is null or length(trim(p_actor_id)) = 0
     or p_actor_role is null or p_actor_role not in ('owner', 'staff') then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Admin order status request is invalid.')::text using errcode = '22023';
  end if;

  select o.* into v_order
  from public.orders as o
  where o.order_number = upper(trim(p_order_ref)) or o.id::text = trim(p_order_ref)
  for update;
  if not found then
    raise exception '%', jsonb_build_object('code', 'ORDER_NOT_FOUND', 'message', 'Order was not found.')::text using errcode = 'P0002';
  end if;

  v_old_status := v_order.status;

  -- Verify transition against canonical transition rules:
  if v_old_status = p_new_status then
    v_allowed := true;
  elsif v_old_status = 'PENDING' and p_new_status in ('CONFIRMED', 'CANCELLED') then
    v_allowed := true;
  elsif v_old_status = 'CONFIRMED' and p_new_status in ('PACKED', 'SHIPPED', 'CANCELLED') then
    v_allowed := true;
  elsif v_old_status = 'PACKED' and p_new_status in ('SHIPPED', 'CANCELLED') then
    v_allowed := true;
  elsif v_old_status = 'SHIPPED' and p_new_status in ('OUT_FOR_DELIVERY', 'DELIVERED', 'CANCELLED') then
    v_allowed := true;
  elsif v_old_status = 'OUT_FOR_DELIVERY' and p_new_status in ('DELIVERED', 'CANCELLED') then
    v_allowed := true;
  elsif v_old_status = 'DELIVERED' and p_new_status in ('RETURN_REQUESTED') then
    v_allowed := true;
  elsif v_old_status = 'RETURN_REQUESTED' and p_new_status in ('RETURNED', 'REFUNDED') then
    v_allowed := true;
  elsif v_old_status = 'RETURNED' and p_new_status in ('REFUNDED') then
    v_allowed := true;
  end if;

  if not v_allowed then
    raise exception '%', jsonb_build_object(
      'code', 'INVALID_STATUS_TRANSITION',
      'message', 'Transition from ' || v_old_status || ' to ' || p_new_status || ' is not allowed.'
    )::text using errcode = 'P0001';
  end if;

  -- If cancelling or returning order, safely restore stock without republishing archived/draft items
  if p_new_status in ('CANCELLED', 'RETURNED') and v_old_status not in ('CANCELLED', 'RETURNED') then
    for v_item in
      select oi.serial_number, sum(oi.quantity)::integer as quantity
      from public.order_items as oi
      where oi.order_id = v_order.id
      group by oi.serial_number
    loop
      select p.id into v_product_id
      from public.products as p
      where p.serial_number = v_item.serial_number
      for update;

      if found then
        update public.products
        set
          stock = stock + v_item.quantity,
          status = case
            when status = 'SOLD_OUT' then 'ACTIVE'
            else status
          end,
          updated_at = timezone('utc'::text, now())
        where id = v_product_id;
      end if;
    end loop;
  end if;

  -- Apply status update (and sync payment_status to REFUNDED if applicable)
  if p_new_status = 'REFUNDED' then
    update public.orders
    set
      status = p_new_status,
      payment_status = 'REFUNDED',
      updated_at = timezone('utc'::text, now())
    where id = v_order.id
    returning * into v_order;
  else
    update public.orders
    set
      status = p_new_status,
      updated_at = timezone('utc'::text, now())
    where id = v_order.id
    returning * into v_order;
  end if;

  -- Append audit log entry
  insert into public.activity_audit_logs (
    actor_id,
    actor_role,
    actor_email,
    table_name,
    record_id,
    action,
    before_state,
    after_state,
    diff,
    ip_address
  )
  values (
    p_actor_id,
    p_actor_role,
    p_actor_email,
    'orders',
    v_order.id::text,
    'STATUS_CHANGE',
    jsonb_build_object('status', v_old_status),
    jsonb_build_object('status', p_new_status, 'payment_status', v_order.payment_status),
    jsonb_build_object('status', jsonb_build_object('from', v_old_status, 'to', p_new_status)),
    p_ip_address
  );

  select coalesce(
    jsonb_agg(
      jsonb_build_object(
        'id', oi.id,
        'order_id', oi.order_id,
        'product_id', oi.product_id,
        'serial_number', oi.serial_number,
        'name', oi.name,
        'price', oi.price,
        'quantity', oi.quantity,
        'size', oi.size,
        'color', oi.color,
        'image_url', oi.image_url
      )
      order by oi.created_at asc
    ),
    '[]'::jsonb
  ) into v_items
  from public.order_items as oi
  where oi.order_id = v_order.id;

  return jsonb_build_object(
    'order', to_jsonb(v_order),
    'order_items', v_items
  );
end;
$$;

declare

  into v_active_orders, v_total_revenue, v_total_orders
  from public.orders;

    'lowStockCount', v_low_stock,

    'totalRevenuePaise', v_total_revenue,
    'totalOrdersCount', v_total_orders
  );

-- ------------------------------------------------------------------------------
-- 5. CRM Customer List & Aggregations RPC (Resolves P1-4 & P1-5)
-- Performs customer and order aggregation in SQL, computing paid-only spend
-- ------------------------------------------------------------------------------
create or replace function public.admin_get_crm_customers(
  p_page integer default 1,
  p_limit integer default 20,
  p_search text default null,
  p_tier text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_offset integer;
  v_search_pattern text;
  v_total bigint := 0;
  v_customers jsonb := '[]'::jsonb;
begin
  if coalesce(auth.role(), '') <> 'service_role' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Admin CRM operation is not authorized.')::text using errcode = '42501';
  end if;

  v_offset := greatest(0, (coalesce(p_page, 1) - 1) * coalesce(p_limit, 20));
  v_search_pattern := case
    when p_search is not null and length(trim(p_search)) > 0 then '%' || lower(trim(p_search)) || '%'
    else null
  end;

  with customer_identities as (
    -- Aggregated order records by customer identity (user_id or lowercase email)
    select
      coalesce(o.customer_id::text, 'email:' || lower(trim(o.customer_email))) as match_key,
      o.customer_id,
      lower(trim(o.customer_email)) as customer_email,
      max(o.customer_name) as latest_customer_name,
      max(o.customer_phone) as latest_customer_phone,
      count(*) filter (where o.status <> 'CANCELLED')::integer as total_orders,
      -- Realized spend counts strictly PAID, non-cancelled, non-refunded orders
      coalesce(sum(o.total_amount) filter (
        where o.payment_status = 'PAID'
          and o.status not in ('CANCELLED', 'REFUNDED')
          and o.payment_status <> 'REFUNDED'
      ), 0)::bigint as total_spend_in_paise,
      min(o.created_at) as first_order_at,
      max(o.created_at) as last_order_at
    from public.orders o
    group by coalesce(o.customer_id::text, 'email:' || lower(trim(o.customer_email))), o.customer_id, lower(trim(o.customer_email))
  ),
  unified_customers as (
    select
      coalesce(c.id::text, 'guest_' || md5(ci.customer_email)) as id,
      coalesce(c.full_name, ci.latest_customer_name, 'Customer') as name,
      coalesce(c.email, ci.customer_email) as email,
      coalesce(c.phone, ci.latest_customer_phone, '') as phone,
      coalesce(ci.total_orders, 0)::integer as "totalOrders",
      coalesce(ci.total_spend_in_paise, 0)::bigint as "totalSpendInPaise",
      ci.first_order_at as "firstOrderAt",
      ci.last_order_at as "lastOrderAt",
      case
        when c.tier is not null and c.tier in ('VIP', 'REGULAR', 'NEW', 'COMMUNITY') then c.tier
        when coalesce(ci.total_spend_in_paise, 0) >= 1000000 then 'VIP'
        when coalesce(ci.total_orders, 0) >= 2 then 'REGULAR'
        else 'NEW'
      end as tier,
      c.staff_notes as notes,
      coalesce(c.saved_addresses, '[]'::jsonb) as "savedAddresses",
      coalesce(c.marketing_opt_in_email, false) as "marketingOptInEmail",
      coalesce(c.marketing_opt_in_whatsapp, false) as "marketingOptInWhatsapp",
      coalesce(c.garments_saved, 0)::integer as "garmentsSaved",
      coalesce(c.co2_avoided_grams, 0)::bigint as "co2AvoidedGrams",
      coalesce(c.water_saved_liters, 0)::bigint as "waterSavedLitres",
      coalesce(c.created_at, ci.first_order_at) as "createdAt"
    from public.customers c
    full outer join customer_identities ci on (
      c.user_id = ci.customer_id or lower(trim(c.email)) = ci.customer_email
    )
  ),
  filtered_customers as (
    select *
    from unified_customers u
    where (
      v_search_pattern is null or
      lower(u.name) like v_search_pattern or
      lower(u.email) like v_search_pattern or
      lower(u.phone) like v_search_pattern or
      lower(coalesce(u.notes, '')) like v_search_pattern
    )
    and (
      p_tier is null or p_tier = 'all' or p_tier = '' or u.tier = p_tier
    )
  )
  select count(*) into v_total from filtered_customers;

  select coalesce(
    jsonb_agg(to_jsonb(fc.*)),
    '[]'::jsonb
  ) into v_customers
  from (
    select *
    from filtered_customers
    order by coalesce("lastOrderAt", "createdAt") desc nulls last
    limit coalesce(p_limit, 20)
    offset v_offset
  ) fc;

  return jsonb_build_object(
    'customers', v_customers,
    'meta', jsonb_build_object(
      'page', coalesce(p_page, 1),
      'limit', coalesce(p_limit, 20),
      'total', v_total,
      'totalPages', case when coalesce(p_limit, 20) > 0 then ceil(v_total::numeric / coalesce(p_limit, 20))::integer else 1 end,
      'hasNext', (v_offset + coalesce(p_limit, 20)) < v_total
    )
  );
end;
$$;

revoke all on function public.admin_get_crm_customers(integer, integer, text, text) from public, anon, authenticated;
grant execute on function public.admin_get_crm_customers(integer, integer, text, text) to service_role;

declare

  v_reset_at timestamptz;
  v_hits integer;
begin
  if p_key is null or length(trim(p_key)) = 0 then
    raise exception 'Rate limit key cannot be empty.';
  end if;

  -- Clean up expired entry if it exists for this key
  delete from public.rate_limit_entries
  where key = p_key and public.rate_limit_entries.reset_at <= v_now;

  -- Insert fresh entry or increment existing hits
  insert into public.rate_limit_entries (key, hits, reset_at)
  values (p_key, 1, v_now + (greatest(coalesce(p_window_ms, 60000), 1000) || ' milliseconds')::interval)
  on conflict (key) do update
  set hits = public.rate_limit_entries.hits + 1
  returning public.rate_limit_entries.hits, public.rate_limit_entries.reset_at into v_hits, v_reset_at;

  return query select v_hits, v_reset_at;
end;
$$;

-- ------------------------------------------------------------------------------
-- 2. Stock Reservation Expiry & Cancellation (Resolves P0-7)
-- Auto-cancels stale unpaid/pending orders and restores stock cleanly
-- ------------------------------------------------------------------------------
create or replace function public.admin_release_expired_reservations(
  p_expiry_hours integer default 24
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_cutoff timestamptz;
  v_order record;
  v_item record;
  v_product_id uuid;
  v_expired_count integer := 0;
  v_restored_count integer := 0;
begin
  if coalesce(auth.role(), '') <> 'service_role' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Cron reservation release is not authorized.')::text using errcode = '42501';
  end if;

  v_cutoff := timezone('utc'::text, now()) - (greatest(coalesce(p_expiry_hours, 24), 1) || ' hours')::interval;

  -- Select all pending, unpaid orders older than expiry window
  for v_order in
    select id, order_number, notes
    from public.orders
    where status = 'PENDING'
      and payment_status = 'PENDING'
      and created_at < v_cutoff
    for update
  loop
    -- Restore inventory for each item in the expired order
    for v_item in
      select oi.serial_number, sum(oi.quantity)::integer as quantity
      from public.order_items as oi
      where oi.order_id = v_order.id
      group by oi.serial_number
    loop
      select p.id into v_product_id
      from public.products as p
      where p.serial_number = v_item.serial_number
      for update;

      if found then
        update public.products
        set
          stock = stock + v_item.quantity,
          status = case
            when status = 'SOLD_OUT' then 'ACTIVE'
            else status
          end,
          updated_at = timezone('utc'::text, now())
        where id = v_product_id;

        v_restored_count := v_restored_count + v_item.quantity;
      end if;
    end loop;

    -- Update order status to CANCELLED with audit note
    update public.orders
    set
      status = 'CANCELLED',
      notes = case
        when v_order.notes is null or length(v_order.notes) = 0 then
          'Auto-cancelled: reservation hold expired after ' || p_expiry_hours || ' hours'
        else
          v_order.notes || ' | Auto-cancelled: reservation hold expired after ' || p_expiry_hours || ' hours'
      end,
      updated_at = timezone('utc'::text, now())
    where id = v_order.id;

    -- Append audit log entry
    insert into public.activity_audit_logs (
      actor_id,
      actor_role,
      actor_email,
      table_name,
      record_id,
      action,
      diff,
      ip_address
    )
    values (
      null,
      'SYSTEM',
      'cron@circleuponline.com',
      'orders',
      v_order.id,
      'EXPIRE_RESERVATION',
      jsonb_build_object(
        'order_number', v_order.order_number,
        'previous_status', 'PENDING',
        'new_status', 'CANCELLED',
        'expiry_hours', p_expiry_hours
      ),
      '127.0.0.1'
    );

    v_expired_count := v_expired_count + 1;
  end loop;

  return jsonb_build_object(
    'expiredOrdersCount', v_expired_count,
    'restoredItemsCount', v_restored_count,
    'expiryHours', p_expiry_hours
  );
end;
$$;

revoke all on function public.admin_release_expired_reservations(integer) from public, anon, authenticated;
grant execute on function public.admin_release_expired_reservations(integer) to service_role;

-- ------------------------------------------------------------------------------
create unique index if not exists idx_event_reg_unique_email
  on public.event_registrations (event_id, lower(email));

create or replace function public.register_for_event_atomic(
  p_event_id uuid,
  p_name text,
  p_email text,
  p_phone text,
  p_notes text default ''
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_event public.events%rowtype;
  v_reg public.event_registrations%rowtype;
  v_clean_email text;
  v_clean_name text;
  v_clean_phone text;
begin
  if p_event_id is null then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Event ID is required.')::text using errcode = '22023';
  end if;

  v_clean_email := lower(trim(coalesce(p_email, '')));
  v_clean_name := trim(coalesce(p_name, ''));
  v_clean_phone := trim(coalesce(p_phone, ''));

  if length(v_clean_name) < 2 or length(v_clean_name) > 100 then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Name must contain 2 to 100 characters.')::text using errcode = '22023';
  end if;
  if v_clean_email !~ '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$' then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Valid email address is required.')::text using errcode = '22023';
  end if;
  if length(v_clean_phone) < 7 or length(v_clean_phone) > 20 then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Valid phone number is required.')::text using errcode = '22023';
  end if;

  -- Acquire exclusive row lock on the event to prevent concurrent registration race conditions
  select * into v_event
  from public.events
  where id = p_event_id
  for update;

  if not found then
    raise exception '%', jsonb_build_object('code', 'EVENT_NOT_FOUND', 'message', 'Event was not found.')::text using errcode = 'P0002';
  end if;

  if not v_event.is_registration_open then
    raise exception '%', jsonb_build_object('code', 'EVENT_REGISTRATION_CLOSED', 'message', 'Registration for this event is closed.')::text using errcode = 'P0001';
  end if;

  if v_event.capacity is not null and v_event.registered_count >= v_event.capacity then
    raise exception '%', jsonb_build_object('code', 'EVENT_CAPACITY_REACHED', 'message', 'This event has reached full capacity.')::text using errcode = 'P0001';
  end if;

  -- Check duplicate registration
  if exists (
    select 1 from public.event_registrations
    where event_id = p_event_id and lower(email) = v_clean_email
  ) then
    raise exception '%', jsonb_build_object('code', 'DUPLICATE_REGISTRATION', 'message', 'A registration for this email has already been received for this event.')::text using errcode = '23505';
  end if;

  -- Insert registration
  insert into public.event_registrations (
    event_id, name, email, phone, notes
  ) values (
    p_event_id, v_clean_name, v_clean_email, v_clean_phone, coalesce(trim(p_notes), '')
  )
  returning * into v_reg;

  -- Increment registered_count atomically
  update public.events
  set registered_count = registered_count + 1,
      updated_at = timezone('utc'::text, now())
  where id = p_event_id;

  return jsonb_build_object(
    'id', v_reg.id,
    'eventId', v_reg.event_id,
    'eventTitle', v_event.title,
    'name', v_reg.name,
    'email', v_reg.email,
    'phone', v_reg.phone,
    'notes', nullif(v_reg.notes, ''),
    'createdAt', v_reg.created_at
  );
end;
$$;

revoke all on function public.register_for_event_atomic(uuid, text, text, text, text) from public, anon, authenticated;
grant execute on function public.register_for_event_atomic(uuid, text, text, text, text) to service_role;

-- ------------------------------------------------------------------------------
-- 4. Admin Refund Processing with Stock Restock
-- ------------------------------------------------------------------------------
create or replace function public.admin_process_order_refund(
  p_order_ref text,
  p_amount_paise bigint,
  p_reason text,
  p_routing_action text,
  p_actor_id text,
  p_actor_email text,
  p_actor_role text,
  p_ip_address text default null,
  p_user_agent text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order public.orders%rowtype;
  v_item record;
  v_refund public.refunds%rowtype;
  v_new_payment_status text;
  v_new_order_status text;
  v_clean_action text;
begin
  if coalesce(auth.role(), '') <> 'service_role' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Admin refund operation is not authorized.')::text using errcode = '42501';
  end if;
  if p_actor_role is null or p_actor_role <> 'owner' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Only owners may process refunds.')::text using errcode = '42501';
  end if;
  if p_amount_paise is null or p_amount_paise <= 0 then
    raise exception '%', jsonb_build_object('code', 'VALIDATION_ERROR', 'message', 'Refund amount must be greater than zero.')::text using errcode = '22023';
  end if;

  v_clean_action := upper(trim(coalesce(p_routing_action, 'RESTOCK')));
  if v_clean_action not in ('RESTOCK', 'DRY_CLEAN', 'ALTERATION', 'DISCARD') then
    v_clean_action := 'RESTOCK';
  end if;

  select o.* into v_order
  from public.orders as o
  where o.order_number = upper(trim(p_order_ref)) or o.id::text = trim(p_order_ref)
  for update;

  if not found then
    raise exception '%', jsonb_build_object('code', 'ORDER_NOT_FOUND', 'message', 'Order was not found.')::text using errcode = 'P0002';
  end if;

  if v_order.payment_status not in ('PAID', 'PARTIALLY_REFUNDED') then
    raise exception '%', jsonb_build_object('code', 'ORDER_NOT_PAID', 'message', 'Cannot refund an unpaid order.')::text using errcode = 'P0001';
  end if;

  if p_amount_paise > v_order.total_paise then
    raise exception '%', jsonb_build_object('code', 'REFUND_EXCEEDS_TOTAL', 'message', 'Refund amount cannot exceed order total.')::text using errcode = '22023';
  end if;

  if p_amount_paise >= v_order.total_paise then
    v_new_payment_status := 'REFUNDED';
    v_new_order_status := 'REFUNDED';
  else
    v_new_payment_status := 'PARTIALLY_REFUNDED';
    v_new_order_status := v_order.status;
  end if;

  -- Insert into public.refunds
  insert into public.refunds (
    order_id, amount_paise, reason, status, approved_by, routing_action
  ) values (
    v_order.id, p_amount_paise, trim(coalesce(p_reason, 'Customer requested refund')), 'APPROVED', p_actor_email, v_clean_action
  )
  returning * into v_refund;

  -- Update order status
  update public.orders
  set payment_status = v_new_payment_status,
      status = v_new_order_status,
      updated_at = timezone('utc'::text, now())
  where id = v_order.id
  returning * into v_order;

  -- If restock action chosen, restore inventory
  if v_clean_action = 'RESTOCK' then
    for v_item in select product_id, quantity from public.order_items where order_id = v_order.id loop
      update public.products
      set stock = stock + v_item.quantity,
          status = case when status = 'SOLD_OUT' then 'ACTIVE' else status end,
          updated_at = timezone('utc'::text, now())
      where id = v_item.product_id;
    end loop;
  end if;

  -- Insert audit log
  insert into public.activity_audit_logs (
    actor_id, actor_email, actor_role, table_name, record_id, action,
    after_state, ip_address, user_agent
  ) values (
    p_actor_id, p_actor_email, p_actor_role, 'Refund', v_refund.id::text,
    'ORDER_REFUNDED',
    jsonb_build_object(
      'orderNumber', v_order.order_number,
      'refundAmountPaise', p_amount_paise,
      'routingAction', v_clean_action,
      'newPaymentStatus', v_new_payment_status
    ),
    coalesce(p_ip_address, '127.0.0.1'), left(coalesce(p_user_agent, 'CircleUp Admin'), 1024)
  );

  return jsonb_build_object(
    'refund', to_jsonb(v_refund),
    'order', to_jsonb(v_order)
  );
end;
$$;

revoke all on function public.admin_process_order_refund(text, bigint, text, text, text, text, text, text, text) from public, anon, authenticated;
grant execute on function public.admin_process_order_refund(text, bigint, text, text, text, text, text, text, text) to service_role;

  if coalesce(auth.role(), '') <> 'service_role' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Admin fulfillment operation is not authorized.')::text using errcode = '42501';
  end if;

  select o.* into v_order
  from public.orders as o
  where o.order_number = upper(trim(p_order_ref)) or o.id::text = trim(p_order_ref)
  for update;

  if not found then
    raise exception '%', jsonb_build_object('code', 'ORDER_NOT_FOUND', 'message', 'Order was not found.')::text using errcode = 'P0002';
  end if;

  insert into public.fulfillments (
    order_id, tracking_number, carrier, status, shipped_at, notes
  ) values (
    v_order.id, trim(p_tracking_number), trim(p_carrier), 'SHIPPED', timezone('utc'::text, now()), nullif(trim(p_notes), '')
  )
  returning * into v_fulfillment;

  -- Advance order status to SHIPPED if it was CONFIRMED or PACKED
  if v_order.status in ('CONFIRMED', 'PACKED') then
    update public.orders
    set status = 'SHIPPED',
        updated_at = timezone('utc'::text, now())
    where id = v_order.id
    returning * into v_order;
  end if;

  -- Insert audit log
  insert into public.activity_audit_logs (
    actor_id, actor_email, actor_role, table_name, record_id, action,
    after_state, ip_address, user_agent
  ) values (
    coalesce(p_actor_id, 'system'), coalesce(p_actor_email, 'admin@circleuponline.com'), coalesce(p_actor_role, 'staff'),
    'Fulfillment', v_fulfillment.id::text, 'ORDER_FULFILLED',
    jsonb_build_object(
      'orderNumber', v_order.order_number,
      'trackingNumber', v_fulfillment.tracking_number,
      'carrier', v_fulfillment.carrier,
      'status', v_fulfillment.status
    ),
    coalesce(p_ip_address, '127.0.0.1'), left(coalesce(p_user_agent, 'CircleUp Admin'), 1024)
  );

  return jsonb_build_object(
    'fulfillment', to_jsonb(v_fulfillment),
    'order', to_jsonb(v_order)
  );
end;
$$;

-- 2. Update process_storefront_checkout to disallow CASH_ON_DELIVERY and support RAZORPAY
create or replace function public.process_storefront_checkout(
  p_key_hash text,
  p_fingerprint text,
  p_customer_id uuid,
  p_customer jsonb,
  p_shipping_address jsonb,
  p_items jsonb,
  p_payment_method text,
  p_promo_code text default null,
  p_notes text default null,
  p_guest_access_token_hash text default null,
  p_guest_access_expires_at timestamptz default null,
  p_order_number text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_existing_idem record;
  v_conflicting_idem record;
  v_claimed_idem record;
  v_order_number text;
  v_order_id uuid;
  v_order_inserted boolean := false;
  v_created_at timestamptz;
  v_whatsapp_number text;
  v_cleaned_whatsapp text;
  v_is_whatsapp_verified boolean;
  v_constraint_name text;
  v_attempt integer;
  v_item record;
  v_serial text;
  v_qty integer;
  v_item_size text;
  v_product record;
  v_missing_serial text;
  v_subtotal bigint := 0;
  v_discount bigint := 0;
  v_delivery_fee bigint := 0;
  v_total_amount bigint := 0;
  v_discounted_subtotal bigint := 0;
  v_coupon record;
  v_coupon_customer_usage bigint := 0;
  v_new_stock integer;
  v_new_status text;
  v_items_json jsonb := '[]'::jsonb;
  v_response_payload jsonb;
begin
  -- ----------------------------------------------------------------------------
  -- 0. PRE-CHECK IDEMPOTENCY KEY
  -- ----------------------------------------------------------------------------
  if p_key_hash is null or length(trim(p_key_hash)) = 0 then
    raise exception '%', jsonb_build_object(
      'code', 'IDEMPOTENCY_KEY_REQUIRED',
      'message', 'A valid Idempotency-Key header is required.'
    )::text;
  end if;

  select *
  into v_existing_idem
  from public.checkout_idempotency_keys
  where key_hash = p_key_hash;

  if found then
    if v_existing_idem.fingerprint <> p_fingerprint then
      raise exception '%', jsonb_build_object(
        'code', 'IDEMPOTENCY_KEY_REUSED_WITH_DIFFERENT_PAYLOAD',
        'message', 'An idempotency key cannot be reused for a different checkout request payload.'
      )::text;
    end if;

    if v_existing_idem.status = 'COMMITTED' then
      return v_existing_idem.response_payload;
    end if;
  end if;

  -- ----------------------------------------------------------------------------
  -- 1. CLAIM IDEMPOTENCY KEY
  -- ----------------------------------------------------------------------------
  insert into public.checkout_idempotency_keys (
    key_hash,
    fingerprint,
    status,
    customer_id,
    order_id,
    created_at,
    expires_at
  )
  values (
    p_key_hash,
    p_fingerprint,
    'PROCESSING',
    p_customer_id,
    null,
    pg_catalog.now(),
    pg_catalog.now() + interval '24 hours'
  )
  on conflict (key_hash) do update
  set
    status = 'PROCESSING',
    expires_at = pg_catalog.now() + interval '24 hours'
  where
    public.checkout_idempotency_keys.key_hash = p_key_hash
    and public.checkout_idempotency_keys.status in ('PENDING', 'FAILED')
  returning * into v_claimed_idem;

  if not found then
    select *
    into v_conflicting_idem
    from public.checkout_idempotency_keys
    where key_hash = p_key_hash;

    if found and v_conflicting_idem.status = 'COMMITTED' then
      if v_conflicting_idem.fingerprint <> p_fingerprint then
        raise exception '%', jsonb_build_object(
          'code', 'IDEMPOTENCY_KEY_REUSED_WITH_DIFFERENT_PAYLOAD',
          'message', 'An idempotency key cannot be reused for a different checkout request payload.'
        )::text;
      end if;
      return v_conflicting_idem.response_payload;
    end if;

    if found and v_conflicting_idem.status = 'PROCESSING' and v_conflicting_idem.created_at > pg_catalog.now() - interval '60 seconds' then
      raise exception '%', jsonb_build_object(
        'code', 'CHECKOUT_ALREADY_IN_PROGRESS',
        'message', 'A checkout transaction with this idempotency key is currently processing. Please wait.'
      )::text;
    end if;

    raise exception '%', jsonb_build_object(
      'code', 'CHECKOUT_RESERVATION_FAILED',
      'message', 'Checkout reservation could not be acquired. Please retry.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 2. VALIDATE PAYMENT METHOD (Razorpay is the sole storefront checkout payment method)
  -- ----------------------------------------------------------------------------
  if p_payment_method = 'CASH_ON_DELIVERY' then
    raise exception '%', jsonb_build_object(
      'code', 'PAYMENT_METHOD_DISALLOWED',
      'message', 'Cash on Delivery is no longer available as a checkout option. Please pay securely with Razorpay.'
    )::text;
  elsif p_payment_method not in ('RAZORPAY', 'ONLINE_UPI', 'WHATSAPP_UPI') then
    raise exception '%', jsonb_build_object(
      'code', 'VALIDATION_ERROR',
      'message', 'Invalid payment method: must be RAZORPAY or ONLINE_UPI.'
    )::text;
  end if;

  if p_payment_method = 'WHATSAPP_UPI' then
    select whatsapp_number, is_whatsapp_verified
    into v_whatsapp_number, v_is_whatsapp_verified
    from public.site_settings
    limit 1;

    v_cleaned_whatsapp := pg_catalog.regexp_replace(coalesce(v_whatsapp_number, ''), '\D', '', 'g');

    if coalesce(v_is_whatsapp_verified, false) is not true or
       v_cleaned_whatsapp is null or
       not (v_cleaned_whatsapp ~ '^[1-9][0-9]{9,14}$') or
       v_cleaned_whatsapp in ('919876543210', '910000000000', '1234567890', '9876543210') or
       v_cleaned_whatsapp like '%000000%' then
      raise exception '%', jsonb_build_object(
        'code', 'PAYMENT_METHOD_UNAVAILABLE',
        'message', 'WhatsApp / UPI payment is temporarily unavailable because store contact details have not been verified.'
      )::text;
    end if;
  end if;

  -- ----------------------------------------------------------------------------
  -- 3. VALIDATE CUSTOMER & SHIPPING ADDRESS
  -- ----------------------------------------------------------------------------
  if p_customer is null or
     trim(coalesce(p_customer->>'name', '')) = '' or
     trim(coalesce(p_customer->>'email', '')) = '' or
     trim(coalesce(p_customer->>'phone', '')) = '' then
    raise exception '%', jsonb_build_object(
      'code', 'MISSING_CUSTOMER_INFO',
      'message', 'Customer name, email, and phone are required.'
    )::text;
  end if;

  if p_shipping_address is null or
     trim(coalesce(p_shipping_address->>'addressLine', '')) = '' or
     trim(coalesce(p_shipping_address->>'city', '')) = '' or
     trim(coalesce(p_shipping_address->>'state', '')) = '' or
     trim(coalesce(p_shipping_address->>'postalCode', '')) = '' then
    raise exception '%', jsonb_build_object(
      'code', 'INCOMPLETE_ADDRESS',
      'message', 'Complete delivery address (line, city, state, postal code) is required.'
    )::text;
  end if;

  if p_items is null or pg_catalog.jsonb_array_length(p_items) = 0 then
    raise exception '%', jsonb_build_object(
      'code', 'EMPTY_ORDER_ITEMS',
      'message', 'Order must contain at least one item.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 4. VALIDATE & RESERVE INVENTORY ATOMICALLY
  -- ----------------------------------------------------------------------------
  for v_item in
    select
      upper(trim(value->>'serialNumber')) as serial_number,
      coalesce((value->>'quantity')::integer, 1) as quantity,
      trim(coalesce(value->>'size', '')) as size
    from pg_catalog.jsonb_array_elements(p_items)
  loop
    v_serial := v_item.serial_number;
    v_qty := v_item.quantity;
    v_item_size := v_item.size;

    if v_serial is null or length(v_serial) = 0 or v_qty <= 0 or v_qty > 10 then
      raise exception '%', jsonb_build_object(
        'code', 'INVALID_ORDER_ITEM',
        'message', 'Each item must have a valid serialNumber and quantity between 1 and 10.'
      )::text;
    end if;

    select *
    into v_product
    from public.products
    where serial_number = v_serial
    for update;

    if not found then
      raise exception '%', jsonb_build_object(
        'code', 'PRODUCT_NOT_FOUND',
        'message', 'Product with serial number ' || v_serial || ' was not found.'
      )::text;
    end if;

    if v_product.status <> 'ACTIVE' or v_product.stock < v_qty then
      raise exception '%', jsonb_build_object(
        'code', 'OUT_OF_STOCK',
        'message', 'Item "' || v_product.name || '" (' || v_serial || ') is no longer available in the requested quantity.'
      )::text;
    end if;

    v_new_stock := v_product.stock - v_qty;
    v_new_status := case when v_new_stock = 0 then 'SOLD_OUT' else 'ACTIVE' end;

    update public.products
    set
      stock = v_new_stock,
      status = v_new_status,
      updated_at = pg_catalog.now()
    where id = v_product.id;

    v_subtotal := v_subtotal + (v_product.price * v_qty);

    v_items_json := v_items_json || pg_catalog.jsonb_build_object(
      'productId', v_product.id,
      'serialNumber', v_product.serial_number,
      'name', v_product.name,
      'price', v_product.price,
      'quantity', v_qty,
      'size', case when v_item_size <> '' then v_item_size else null end,
      'slug', v_product.slug,
      'image', case when pg_catalog.jsonb_array_length(v_product.images) > 0 then v_product.images->>0 else null end
    );
  end loop;

  -- ----------------------------------------------------------------------------
  -- 5. COUPON & PROMOTION EVALUATION
  -- ----------------------------------------------------------------------------
  if p_promo_code is not null and trim(p_promo_code) <> '' then
    select *
    into v_coupon
    from public.coupons
    where code = upper(trim(p_promo_code))
      and is_active = true
      and starts_at <= pg_catalog.now()
      and (expires_at is null or expires_at > pg_catalog.now())
    for update;

    if found then
      if (v_coupon.usage_limit is null or v_coupon.usage_count < v_coupon.usage_limit)
         and (v_coupon.min_order_amount is null or v_subtotal >= v_coupon.min_order_amount) then

        if p_customer_id is not null and v_coupon.per_customer_limit is not null and v_coupon.per_customer_limit > 0 then
          select count(*)
          into v_coupon_customer_usage
          from public.orders
          where customer_id = p_customer_id
            and promo_code = v_coupon.code
            and status <> 'CANCELLED';

          if v_coupon_customer_usage >= v_coupon.per_customer_limit then
            raise exception '%', jsonb_build_object(
              'code', 'COUPON_LIMIT_REACHED',
              'message', 'You have reached the maximum number of redemptions for promo code "' || v_coupon.code || '".'
            )::text;
          end if;
        end if;

        if v_coupon.discount_type = 'PERCENTAGE' then
          v_discount := (v_subtotal * v_coupon.discount_value) / 10000;
          if v_coupon.max_discount_amount is not null and v_discount > v_coupon.max_discount_amount then
            v_discount := v_coupon.max_discount_amount;
          end if;
        elsif v_coupon.discount_type = 'FIXED_AMOUNT' then
          v_discount := v_coupon.discount_value;
          if v_discount > v_subtotal then
            v_discount := v_subtotal;
          end if;
        end if;

        update public.coupons
        set
          usage_count = usage_count + 1,
          updated_at = pg_catalog.now()
        where id = v_coupon.id;
      end if;
    end if;
  end if;

  v_discounted_subtotal := v_subtotal - v_discount;
  if v_discounted_subtotal < 0 then
    v_discounted_subtotal := 0;
  end if;

  if v_discounted_subtotal >= 150000 then
    v_delivery_fee := 0;
  else
    v_delivery_fee := 9000;
  end if;

  v_total_amount := v_discounted_subtotal + v_delivery_fee;

  -- ----------------------------------------------------------------------------
  -- 6. GENERATE COLLISION-RESISTANT ORDER NUMBER & INSERT ORDER
  -- ----------------------------------------------------------------------------
  for v_attempt in 1..5 loop
    if p_order_number is not null and length(trim(p_order_number)) > 0 and v_attempt = 1 then
      v_order_number := upper(trim(p_order_number));
    else
      v_order_number := 'ORD-' || to_char(timezone('Asia/Kolkata', pg_catalog.now()), 'YYMMDD') || '-' ||
        upper(replace(gen_random_uuid()::text, '-', ''));
    end if;

    begin
      insert into public.orders (
        order_number,
        customer_id,
        customer_name,
        customer_email,
        customer_phone,
        total_amount,
        subtotal,
        discount_amount,
        shipping_amount,
        status,
        payment_status,
        payment_method,
        shipping_address,
        promo_code,
        notes,
        guest_access_token_hash,
        guest_access_expires_at,
        created_at,
        updated_at
      )
      values (
        v_order_number,
        p_customer_id,
        trim(p_customer->>'name'),
        lower(trim(p_customer->>'email')),
        trim(p_customer->>'phone'),
        v_total_amount,
        v_subtotal,
        v_discount,
        v_delivery_fee,
        case when p_payment_method in ('RAZORPAY', 'ONLINE_UPI') then 'PENDING' else 'CONFIRMED' end,
        'PENDING',
        p_payment_method,
        p_shipping_address,
        case when p_promo_code is not null and trim(p_promo_code) != '' then upper(trim(p_promo_code)) else null end,
        p_notes,
        p_guest_access_token_hash,
        p_guest_access_expires_at,
        pg_catalog.now(),
        pg_catalog.now()
      )
      returning id, created_at into v_order_id, v_created_at;

      v_order_inserted := true;
      exit;
    exception when unique_violation then
      get stacked diagnostics v_constraint_name = constraint_name;
      if v_constraint_name = 'orders_order_number_key' then
        continue;
      else
        raise;
      end if;
    end;
  end loop;

  if not v_order_inserted then
    raise exception '%', jsonb_build_object(
      'code', 'ORDER_NUMBER_GENERATION_FAILED',
      'message', 'Failed to allocate unique order number after multiple attempts.'
    )::text;
  end if;

  -- ----------------------------------------------------------------------------
  -- 7. INSERT ORDER ITEMS
  -- ----------------------------------------------------------------------------
  for v_item in
    select
      (value->>'productId')::uuid as product_id,
      value->>'serialNumber' as serial_number,
      value->>'name' as name,
      (value->>'price')::bigint as price,
      (value->>'quantity')::integer as quantity,
      value->>'size' as size,
      value->>'image' as image_url,
      value->>'slug' as slug
    from pg_catalog.jsonb_array_elements(v_items_json)
  loop
    insert into public.order_items (
      order_id,
      product_id,
      serial_number,
      name,
      price,
      quantity,
      size,
      image_url,
      created_at
    )
    values (
      v_order_id,
      v_item.product_id,
      v_item.serial_number,
      v_item.name,
      v_item.price,
      v_item.quantity,
      v_item.size,
      v_item.image_url,
      pg_catalog.now()
    );
  end loop;

  -- ----------------------------------------------------------------------------
  -- 8. CONSTRUCT AUTHORITATIVE ORDER RESPONSE PAYLOAD
  -- ----------------------------------------------------------------------------
  v_response_payload := jsonb_build_object(
    'id', v_order_id,
    'orderNumber', v_order_number,
    'customerId', p_customer_id,
    'customer', jsonb_build_object(
      'name', trim(p_customer->>'name'),
      'email', lower(trim(p_customer->>'email')),
      'phone', trim(p_customer->>'phone')
    ),
    'shippingAddress', p_shipping_address,
    'items', v_items_json,
    'subtotal', v_subtotal,
    'discount', v_discount,
    'deliveryFee', v_delivery_fee,
    'totalAmount', v_total_amount,
    'status', case when p_payment_method in ('RAZORPAY', 'ONLINE_UPI') then 'PENDING' else 'CONFIRMED' end,
    'paymentMethod', p_payment_method,
    'paymentStatus', 'PENDING',
    'promoCode', case when p_promo_code is not null and trim(p_promo_code) != '' then upper(trim(p_promo_code)) else null end,
    'notes', p_notes,
    'createdAt', v_created_at
  );

  -- ----------------------------------------------------------------------------
  -- 9. COMMIT IDEMPOTENCY RECORD
  -- ----------------------------------------------------------------------------
  update public.checkout_idempotency_keys
  set
    status = 'COMMITTED',
    order_id = v_order_id,
    response_payload = v_response_payload,
    expires_at = pg_catalog.now() + interval '7 days'
  where key_hash = p_key_hash;

  return v_response_payload;
end;
$$;

-- 3. Create RPC public.storefront_verify_razorpay_payment
create or replace function public.storefront_verify_razorpay_payment(
  p_order_number text,
  p_razorpay_order_id text,
  p_razorpay_payment_id text,
  p_razorpay_signature text
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order record;
  v_items jsonb;
begin
  if p_order_number is null or length(trim(p_order_number)) = 0 then
    raise exception '%', jsonb_build_object(
      'code', 'VALIDATION_ERROR',
      'message', 'A valid order number is required for payment verification.'
    )::text;
  end if;

  select *
  into v_order
  from public.orders
  where order_number = upper(trim(p_order_number))
  for update;

  if not found then
    raise exception '%', jsonb_build_object(
      'code', 'ORDER_NOT_FOUND',
      'message', 'Order ' || p_order_number || ' was not found.'
    )::text;
  end if;

  -- Idempotent return if already marked PAID
  if v_order.payment_status = 'PAID' then
    select coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', oi.id,
          'productId', oi.product_id,
          'serialNumber', oi.serial_number,
          'name', oi.name,
          'price', oi.price,
          'quantity', oi.quantity,
          'size', oi.size,
          'image', oi.image_url
        )
        order by oi.created_at asc
      ),
      '[]'::jsonb
    ) into v_items
    from public.order_items as oi
    where oi.order_id = v_order.id;

    return jsonb_build_object(
      'order', to_jsonb(v_order),
      'items', v_items
    );
  end if;

  if v_order.status = 'CANCELLED' then
    raise exception '%', jsonb_build_object(
      'code', 'ORDER_ALREADY_CANCELLED',
      'message', 'Cannot verify payment for a cancelled order.'
    )::text;
  end if;

  -- Update order to PAID and advance to CONFIRMED
  update public.orders
  set
    payment_status = 'PAID',
    status = case when status = 'PENDING' then 'CONFIRMED' else status end,
    payment_id = p_razorpay_payment_id,
    razorpay_order_id = p_razorpay_order_id,
    razorpay_payment_id = p_razorpay_payment_id,
    razorpay_signature = p_razorpay_signature,
    updated_at = timezone('utc'::text, now())
  where id = v_order.id
  returning * into v_order;

  -- Append audit log
  insert into public.activity_audit_logs (
    actor_id,
    actor_role,
    actor_email,
    table_name,
    record_id,
    action,
    before_state,
    after_state,
    diff
  )
  values (
    coalesce(v_order.customer_id::text, 'guest'),
    'system',
    v_order.customer_email,
    'orders',
    v_order.id::text,
    'ORDER_PAYMENT_VERIFIED',
    jsonb_build_object('payment_status', 'PENDING', 'status', 'PENDING'),
    jsonb_build_object('payment_status', 'PAID', 'status', v_order.status, 'payment_id', p_razorpay_payment_id),
    jsonb_build_object('razorpay_payment_id', p_razorpay_payment_id, 'razorpay_order_id', p_razorpay_order_id)
  );

  select coalesce(
    jsonb_agg(
      jsonb_build_object(
        'id', oi.id,
        'productId', oi.product_id,
        'serialNumber', oi.serial_number,
        'name', oi.name,
        'price', oi.price,
        'quantity', oi.quantity,
        'size', oi.size,
        'image', oi.image_url
      )
      order by oi.created_at asc
    ),
    '[]'::jsonb
  ) into v_items
  from public.order_items as oi
  where oi.order_id = v_order.id;

  return jsonb_build_object(
    'order', to_jsonb(v_order),
    'items', v_items
  );
end;
$$;

-- 4. Create RPC public.storefront_cancel_pending_order to release inventory cleanly
create or replace function public.storefront_cancel_pending_order(
  p_order_number text,
  p_reason text default 'Checkout dismissed or payment failed'
)
returns boolean
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_order record;
  v_item record;
  v_product_id uuid;
begin
  if p_order_number is null or length(trim(p_order_number)) = 0 then
    raise exception '%', jsonb_build_object(
      'code', 'VALIDATION_ERROR',
      'message', 'A valid order number is required.'
    )::text;
  end if;

  select *
  into v_order
  from public.orders
  where order_number = upper(trim(p_order_number))
  for update;

  if not found then
    raise exception '%', jsonb_build_object(
      'code', 'ORDER_NOT_FOUND',
      'message', 'Order ' || p_order_number || ' was not found.'
    )::text;
  end if;

  -- Idempotent return if already CANCELLED
  if v_order.status = 'CANCELLED' then
    return true;
  end if;

  if v_order.payment_status = 'PAID' then
    raise exception '%', jsonb_build_object(
      'code', 'CANNOT_CANCEL_PAID_ORDER',
      'message', 'Paid orders cannot be automatically cancelled via storefront dismissal.'
    )::text;
  end if;

  -- Restore stock for each item in the order
  for v_item in
    select oi.serial_number, sum(oi.quantity)::integer as quantity
    from public.order_items as oi
    where oi.order_id = v_order.id
    group by oi.serial_number
  loop
    select p.id into v_product_id
    from public.products as p
    where p.serial_number = v_item.serial_number
    for update;

    if found then
      update public.products
      set
        stock = stock + v_item.quantity,
        status = case
          when status = 'SOLD_OUT' then 'ACTIVE'
          else status
        end,
        updated_at = timezone('utc'::text, now())
      where id = v_product_id;
    end if;
  end loop;

  -- Mark order cancelled with note
  update public.orders
  set
    status = 'CANCELLED',
    notes = coalesce(notes, '') || ' | ' || coalesce(p_reason, 'Checkout dismissed'),
    updated_at = timezone('utc'::text, now())
  where id = v_order.id;

  -- Audit log
  insert into public.activity_audit_logs (
    actor_id,
    actor_role,
    actor_email,
    table_name,
    record_id,
    action,
    before_state,
    after_state,
    diff
  )
  values (
    coalesce(v_order.customer_id::text, 'guest'),
    'system',
    v_order.customer_email,
    'orders',
    v_order.id::text,
    'ORDER_STATUS_UPDATE',
    jsonb_build_object('status', v_order.status, 'payment_status', v_order.payment_status),
    jsonb_build_object('status', 'CANCELLED', 'payment_status', v_order.payment_status),
    jsonb_build_object('reason', p_reason, 'operation', 'ORDER_RESERVATION_CANCELLED')
  );

  return true;
end;
$$;

revoke all on function public.storefront_cancel_pending_order(text, text) from public;
grant execute on function public.storefront_cancel_pending_order(text, text) to service_role;

-- 3. Fix admin_get_crm_customers join condition
create or replace function public.admin_get_crm_customers(
  p_page integer default 1,
  p_limit integer default 20,
  p_search text default null,
  p_tier text default null
)
returns jsonb
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_offset integer;
  v_search_pattern text;
  v_total bigint := 0;
  v_customers jsonb := '[]'::jsonb;
begin
  if coalesce(auth.role(), '') <> 'service_role' then
    raise exception '%', jsonb_build_object('code', 'FORBIDDEN', 'message', 'Admin CRM operation is not authorized.')::text using errcode = '42501';
  end if;

  v_offset := greatest(0, (coalesce(p_page, 1) - 1) * coalesce(p_limit, 20));
  v_search_pattern := case
    when p_search is not null and length(trim(p_search)) > 0 then '%' || lower(trim(p_search)) || '%'
    else null
  end;

  with customer_identities as (
    select
      lower(trim(o.customer_email)) as customer_email,
      max(o.customer_name) as latest_customer_name,
      max(o.customer_phone) as latest_customer_phone,
      count(*) filter (where o.status <> 'CANCELLED')::integer as total_orders,
      coalesce(sum(o.total_amount) filter (
        where o.payment_status = 'PAID'
          and o.status not in ('CANCELLED', 'REFUNDED')
          and o.payment_status <> 'REFUNDED'
      ), 0)::bigint as total_spend_in_paise,
      min(o.created_at) as first_order_at,
      max(o.created_at) as last_order_at
    from public.orders o
    where o.customer_email is not null and length(trim(o.customer_email)) > 0
    group by lower(trim(o.customer_email))
  ),
  unified_customers as (
    select
      coalesce(c.id::text, 'guest_' || md5(ci.customer_email)) as id,
      coalesce(c.full_name, ci.latest_customer_name, 'Customer') as name,
      coalesce(lower(trim(c.email)), ci.customer_email) as email,
      coalesce(c.phone, ci.latest_customer_phone, '') as phone,
      coalesce(ci.total_orders, 0)::integer as "totalOrders",
      coalesce(ci.total_spend_in_paise, 0)::bigint as "totalSpendInPaise",
      ci.first_order_at as "firstOrderAt",
      ci.last_order_at as "lastOrderAt",
      case
        when c.tier is not null and c.tier in ('VIP', 'REGULAR', 'NEW', 'COMMUNITY') then c.tier
        when coalesce(ci.total_spend_in_paise, 0) >= 1000000 then 'VIP'
        when coalesce(ci.total_orders, 0) >= 2 then 'REGULAR'
        else 'NEW'
      end as tier,
      c.staff_notes as notes,
      coalesce(c.saved_addresses, '[]'::jsonb) as "savedAddresses",
      coalesce(c.marketing_opt_in_email, false) as "marketingOptInEmail",
      coalesce(c.marketing_opt_in_whatsapp, false) as "marketingOptInWhatsapp",
      coalesce(c.garments_saved, 0)::integer as "garmentsSaved",
      coalesce(c.co2_avoided_grams, 0)::bigint as "co2AvoidedGrams",
      coalesce(c.water_saved_liters, 0)::bigint as "waterSavedLitres",
      coalesce(c.created_at, ci.first_order_at) as "createdAt"
    from public.customers c
    full outer join customer_identities ci on lower(trim(c.email)) = ci.customer_email
  ),
  filtered_customers as (
    select *
    from unified_customers u
    where (
      v_search_pattern is null or
      lower(u.name) like v_search_pattern or
      lower(u.email) like v_search_pattern or
      lower(u.phone) like v_search_pattern or
      lower(coalesce(u.notes, '')) like v_search_pattern
    )
    and (
      p_tier is null or p_tier = 'all' or p_tier = '' or u.tier = p_tier
    )
  )
  select count(*) into v_total from filtered_customers;

  select coalesce(
    jsonb_agg(to_jsonb(fc.*)),
    '[]'::jsonb
  ) into v_customers
  from (
    select *
    from filtered_customers
    order by coalesce("lastOrderAt", "createdAt") desc nulls last
    limit coalesce(p_limit, 20)
    offset v_offset
  ) fc;

  return jsonb_build_object(
    'customers', v_customers,
    'meta', jsonb_build_object(
      'page', coalesce(p_page, 1),
      'limit', coalesce(p_limit, 20),
      'total', v_total,
      'totalPages', case when coalesce(p_limit, 20) > 0 then ceil(v_total::numeric / coalesce(p_limit, 20))::integer else 1 end,
      'hasNext', (v_offset + coalesce(p_limit, 20)) < v_total
    )
  );
end;
$$;

revoke all on function public.admin_get_crm_customers(integer, integer, text, text) from public, anon, authenticated;
grant execute on function public.admin_get_crm_customers(integer, integer, text, text) to service_role;
